     1→/**
     2→ * Truth table: auto-generated for the selected input width, fully editable
     3→ * (click an output cell to cycle 0 → 1 → X, or click a segment directly on the
     4→ * row's preview glyph). Clicking a row selects that input.
     5→ *
     6→ * Features:
     7→ *  - Live mini 7-segment preview of every row (derived from the table itself),
     8→ *    shown after the Hex column; its segments are clickable to toggle cells.
     9→ *  - Drag a row's preview glyph onto another row to copy its output pattern —
    10→ *    inputs are excluded, they always stay untouched.
    11→ *  - Input variables default to A B C D / A B C D E (A = MSB) and can be
    12→ *    renamed by clicking a variable header — names flow through expressions,
    13→ *    K-map, circuit and every export.
    14→ *  - Row order can start from any code (e.g. ABCD = 0001, wrapping around).
    15→ *  - Shift tools: rotate one row's values, one segment's column, or the whole
    16→ *    table (all rows / all columns) — with one-level undo.
    17→ *  - FONT preset window controls (which glyph starts the mapping).
    18→ */

    41→function cellColor(v: SegmentOutput): string {
    42→  return v === 1
    43→    ? 'border-emerald-500/40 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400'
    44→    : v === 'X'

    49→/**
    50→ * One memoized truth-table row. Extracted from the map so drag feedback
    51→ * (dropRow changes) and other parent re-renders never re-render the other
    52→ * 15–31 rows: every prop is a primitive, a stable ref from the immutable
    53→ * store, or a useCallback handler, so React.memo's shallow compare holds.
    54→ *
    55→ * Keyboard support: the row itself is focusable (Tab) — Enter/Space select it,
    56→ * "c" copies its segment outputs to an internal clipboard and "v" pastes the
    57→ * clipboard onto the focused row (the keyboard twin of drag-to-copy).

    59→interface TruthRowProps {

    61→  inputBits: InputWidth;
    62→  /** Character label for the Char column (font glyph / hex digit / null). */

    75→  onRowHover: (input: number) => void;
    76→  onRowLeave: (input: number) => void;
    77→  /** Key handler for the focused row (select / copy / paste). */
    78→  onRowKeyDown: (e: React.KeyboardEvent<HTMLTableRowElement>, input: number) => void;

    81→const TruthRow = memo(function TruthRow({
    82→  row, inputBits, char, active, isDragSource, isDropTarget,
    83→  registerRef, onSelectRow, onCycleCell, onTogglePreviewSegment,
    84→  onDragRowStart, onDragRowEnd, onRowDrop, onRowHover, onRowLeave, onRowKeyDown,
    85→}: TruthRowProps) {
    86→  const bits = row.input.toString(2).padStart(inputBits, '0');
    87→  return (
    88→    <TableRow
    89→      ref={(el) => {
    90→        registerRef(row.input, el);
    91→      }}
    92→      tabIndex={0}
    93→      className={cn(
    94→        'cursor-pointer',
    95→        'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-inset focus-visible:ring-ring',
    96→        active ? 'bg-primary/10' : 'hover:bg-muted/50',
    97→        isDragSource && 'opacity-50',
    98→        isDropTarget && 'bg-primary/15 ring-2 ring-inset ring-primary/50',
    99→      )}
   100→      onClick={() => onSelectRow(row.input)}
   101→      onKeyDown={(e) => onRowKeyDown(e, row.input)}
   102→      aria-label={`Truth table row m${row.input} (${bits}) — Enter selects, C copies outputs, V pastes`}
   103→      aria-keyshortcuts="Enter Space c v"
   104→      onDragOver={(e) => {
   105→        if (!e.dataTransfer.types.includes('text/x-sevenseg-row')) return;
   106→        e.preventDefault();
   107→        e.dataTransfer.dropEffect = 'copy';
   108→        onRowHover(row.input);
   109→      }}
   110→      onDragLeave={(e) => {
   111→        if (!e.currentTarget.contains(e.relatedTarget as Node | null)) {
   112→          onRowLeave(row.input);
   113→        }
   114→      }}
   115→      onDrop={(e) => onRowDrop(e, row.input)}
   116→    >
   117→      {[...bits].map((b, i) => (
   118→        <TableCell key={i} className="px-2 py-1 text-center font-mono text-xs tabular-nums">
   119→          {b}
   120→        </TableCell>
   121→      ))}
   122→      <TableCell className="px-2 py-1 text-center font-mono text-xs tabular-nums">
   123→        {row.input}
   124→      </TableCell>
   125→      <TableCell className="px-2 py-1 text-center font-mono text-xs">
   126→        {formatHex(row.input)}
   127→      </TableCell>
   128→      <TableCell className="border-r border-border/70 px-2 py-1 text-center font-mono text-xs">
   129→        {char === null ? '' : char === ' ' ? '␠' : char}
   130→      </TableCell>
   131→      <TableCell className="px-1 py-0.5 text-center">
   132→        <div
   133→          className="group relative flex cursor-grab flex-col items-center leading-none active:cursor-grabbing"
   134→          draggable
   135→          onDragStart={(e) => {
   136→            e.dataTransfer.setData('text/x-sevenseg-row', String(row.input));
   137→            e.dataTransfer.effectAllowed = 'copy';
   138→            onDragRowStart(row.input);
   139→          }}
   140→          onDragEnd={() => {
   141→            onDragRowEnd();
   142→          }}
   143→          title={`Pattern of m${row.input} — click a segment to toggle it (0 ↔ 1); drag onto another row to copy this pattern (inputs are excluded and stay unchanged)`}
   144→        >
   145→          <GlyphPreview
   146→            outputs={row.outputs}
   147→            interactive
   148→            onSegmentClick={(seg) => {
   149→              onTogglePreviewSegment(row.input, seg);
   150→            }}
   151→            className="relative z-0 h-10 origin-center transition-transform duration-150 ease-out group-hover:z-20 group-hover:scale-[1.65]"
   152→            label={`Pattern of row ${row.input}${char ? ` (${char})` : ''} — segments clickable`}
   153→          />
   154→        </div>
   155→      </TableCell>
   156→      {SEGMENT_NAMES.map((seg) => {
   157→        const v = row.outputs[seg];
   158→        return (
   159→          <TableCell
   160→            key={seg}
   161→            className={cn('px-1 py-1 text-center', seg === 'DP' && 'border-l border-border/70')}
   162→          >
   163→            <button
   164→              type="button"
   165→              className={cn(
   166→                'inline-flex h-7 w-7 items-center justify-center rounded border font-mono text-xs font-semibold transition-all hover:scale-110 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring',
   167→                cellColor(v),
   168→              )}
   169→              onClick={(e) => {
   170→                e.stopPropagation();
   171→                onCycleCell(row.input, seg);
   172→              }}
   173→              aria-label={`Row ${row.input} segment ${seg}: ${v}, click to cycle`}
   174→              title={`m${row.input} · ${seg} = ${v} — click to cycle`}
   175→            >
   176→              {v}
   177→            </button>
   178→          </TableCell>
   179→        );
   180→      })}
   181→    </TableRow>
   182→  );
   183→});
