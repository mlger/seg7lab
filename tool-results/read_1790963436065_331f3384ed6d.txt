     1→/**
     2→ * Logic-gate circuit visualizer, generated from the minimized Boolean
     3→ * expression of the selected segment.
     4→ *
     5→ * Features: live signal simulation (active paths glow), zoom (wheel/buttons),
     6→ * pan (drag background), draggable gates, click-to-inspect gates, clickable
     7→ * input rails that drive the whole simulator, and an interactive gate
     8→ * reference (AND OR NOT NAND NOR XOR XNOR).
     9→ */

    14→  CircuitBoard, ZoomIn, ZoomOut, Maximize, RotateCcw, MousePointerClick, Lightbulb,
    15→} from 'lucide-react';

    25→
    26→type GateType = 'input' | 'not' | 'and' | 'or' | 'xor' | 'nand' | 'nor' | 'xnor' | 'const' | 'out';
    27→
    28→interface CNode {
    29→  id: string;
    30→  type: GateType;
    31→  label: string;
    32→  varName?: string;
    33→  constValue?: 0 | 1;
    34→  operandIds: string[];
    35→  x: number;
    36→  y: number;
    37→  w: number;
    38→  h: number;
    39→}
    40→
    41→interface CWire {
    42→  id: string;
    43→  fromId: string;
    44→  toId: string;
    45→  pin: number;
    46→}
    47→
    48→const GATE_INFO: Record<string, { desc: string; expr: string }> = {
    49→  AND: { desc: 'Outputs 1 only when every input is 1.', expr: 'Y = A·B' },
    50→  OR: { desc: 'Outputs 1 when at least one input is 1.', expr: 'Y = A + B' },
    51→  NOT: { desc: 'Inverts its single input.', expr: "Y = A'" },
    52→  NAND: { desc: 'AND followed by inversion — outputs 0 only when every input is 1.', expr: "Y = (A·B)'" },
    53→  NOR: { desc: 'OR followed by inversion — outputs 1 only when every input is 0.', expr: "Y = (A+B)'" },
    54→  XOR: { desc: 'Outputs 1 when an odd number of inputs are 1.', expr: 'Y = A ⊕ B' },
    55→  XNOR: { desc: 'XOR followed by inversion — outputs 1 when inputs are equal.', expr: "Y = (A ⊕ B)'" },
    56→};
    57→
    58→/* ------------------------------------------------------------------ */
    59→/* Netlist construction                                                */
    60→/* ------------------------------------------------------------------ */
    61→
    62→function buildCircuit(terms: Term[], form: 'sop' | 'pos', varNames: string[]) {
    63→  const nodes: CNode[] = [];
    64→  const byId = new Map<string, CNode>();
    65→  const add = (n: CNode) => {
    66→    nodes.push(n);
    67→    byId.set(n.id, n);
    68→    return n;
    69→  };
    70→
    71→  const inputNodes = new Map<string, CNode>();
    72→  varNames.forEach((v, i) => {
    73→    const n = add({
    74→      id: `in-${v}`, type: 'input', label: v, varName: v, operandIds: [],
    75→      x: 56, y: 48 + i * 76, w: 64, h: 44,
    76→    });
    77→    inputNodes.set(v, n);
    78→  });
    79→
    80→  const notNodes = new Map<string, CNode>();
    81→  const literalNode = (lit: Literal): CNode => {
    82→    const base = inputNodes.get(lit.name)!;
    83→    if (!lit.complemented) return base;
    84→    let not = notNodes.get(lit.name);
    85→    if (!not) {
    86→      not = add({
    87→        id: `not-${lit.name}`, type: 'not', label: `NOT ${lit.name}`,
    88→        operandIds: [base.id], x: 168, y: base.y + 4, w: 60, h: 36,
    89→      });
    90→      notNodes.set(lit.name, not);
    91→    }
    92→    return not;
    93→  };
    94→
    95→  const literalNodes = terms.map((t) => t.literals.map(literalNode));
    96→  const termGate = form === 'sop' ? 'and' : 'or';
    97→  const rootGate = form === 'sop' ? 'or' : 'and';
    98→
    99→  let yCursor = 52;
   100→  const termNodes: CNode[] = [];
   101→  terms.forEach((t, ti) => {
   102→    const pins = literalNodes[ti];
   103→    if (pins.length <= 1) {
   104→      termNodes.push(pins[0] ?? add({
   105→        id: `const-${ti}`, type: 'const', label: '1', constValue: 1, operandIds: [],
   106→        x: 300, y: yCursor, w: 48, h: 36,
   107→      }));
   108→    } else {
   109→      const h = Math.max(48, pins.length * 26 + 18);
   110→      termNodes.push(add({
   111→        id: `term-${ti}`, type: termGate, label: `${termGate.toUpperCase()}`,
   112→        operandIds: pins.map((p) => p.id), x: 316, y: yCursor, w: 76, h,
   113→      }));
   114→      yCursor += h + 40;
   115→    }
   116→  });
   117→
   118→  const distinct = termNodes.filter((n, i, arr) => arr.findIndex((m) => m.id === n.id) === i);
   119→  let root: CNode;
   120→  if (terms.length === 0) {
   121→    root = add({
   122→      id: 'const-root', type: 'const', label: form === 'sop' ? '0' : '1',
   123→      constValue: form === 'sop' ? 0 : 1, operandIds: [], x: 560, y: 120, w: 48, h: 36,
   124→    });
   125→  } else if (distinct.length === 1) {
   126→    root = distinct[0];
   127→  } else {
   128→    const h = Math.max(48, distinct.length * 26 + 18);
   129→    const rootY = 52 + (yCursor - 92) / 2 - h / 2;
   130→    root = add({
   131→      id: 'root', type: rootGate, label: rootGate.toUpperCase(),
   132→      operandIds: distinct.map((n) => n.id), x: 560, y: Math.max(40, rootY), w: 76, h,
   133→    });
   134→  }
   135→
   136→  const out = add({
   137→    id: 'out', type: 'out', label: 'OUT', operandIds: [root.id],
   138→    x: 730, y: root.y + root.h / 2 - 34, w: 120, h: 68,
   139→  });
   140→
   141→  const wires: CWire[] = [];
   142→  for (const n of nodes) {
   143→    n.operandIds.forEach((fromId, pin) => {
   144→      wires.push({ id: `${fromId}->${n.id}:${pin}`, fromId, toId: n.id, pin });
   145→    });
   146→  }
   147→
   148→  const maxX = Math.max(...nodes.map((n) => n.x + n.w)) + 40;
   149→  const maxY = Math.max(...nodes.map((n) => n.y + n.h)) + 40;
   150→  return { nodes, wires, byId, width: Math.max(920, maxX), height: Math.max(360, maxY) };
   151→}

   154→/* Gate symbols                                                        */
   155→/* ------------------------------------------------------------------ */
   156→
   157→function gateSymbol(type: GateType, x: number, y: number, w: number, h: number, active: boolean) {
   158→  const stroke = active ? '#10b981' : 'currentColor';
   159→  const fill = active ? 'rgba(16,185,129,0.12)' : 'rgba(128,128,128,0.07)';
   160→  const bubble = (bx: number) => (
   161→    <circle cx={bx} cy={y + h / 2} r={6} fill="none" stroke={stroke} strokeWidth={2} />
   162→  );
   163→  switch (type) {
   164→    case 'and':
   165→    case 'nand': {
   166→      const bodyW = type === 'nand' ? w - 12 : w;
   167→      return (
   168→        <>
   169→          <path
   170→            d={`M ${x},${y} L ${x + bodyW * 0.52},${y} A ${h / 2},${h / 2} 0 0 1 ${x + bodyW * 0.52},${y + h} L ${x},${y + h} Z`}
   171→            fill={fill} stroke={stroke} strokeWidth={2}
   172→          />
   173→          {type === 'nand' && bubble(x + w - 6)}
   174→        </>
   175→      );
   176→    }
   177→    case 'or':
   178→    case 'nor': {
   179→      const bodyW = type === 'nor' ? w - 12 : w;
   180→      const mx = x + bodyW;
   181→      return (
   182→        <>
   183→          <path
   184→            d={`M ${x},${y} Q ${x + bodyW * 0.55},${y} ${mx},${y + h / 2} Q ${x + bodyW * 0.55},${y + h} ${x},${y + h} Q ${x + bodyW * 0.34},${y + h / 2} ${x},${y} Z`}
   185→            fill={fill} stroke={stroke} strokeWidth={2}
   186→          />
   187→          {type === 'nor' && bubble(x + w - 6)}
   188→        </>
   189→      );
   190→    }
   191→    case 'xor':
   192→    case 'xnor': {
   193→      const bodyW = type === 'xnor' ? w - 12 : w;
   194→      const mx = x + bodyW;
   195→      return (
   196→        <>
   197→          <path
   198→            d={`M ${x + 8},${y} Q ${x + bodyW * 0.55},${y} ${mx},${y + h / 2} Q ${x + bodyW * 0.55},${y + h} ${x + 8},${y + h} Q ${x + bodyW * 0.34},${y + h / 2} ${x + 8},${y} Z`}
   199→            fill={fill} stroke={stroke} strokeWidth={2}
   200→          />
   201→          <path
   202→            d={`M ${x},${y} Q ${x + bodyW * 0.34},${y + h / 2} ${x},${y + h}`}
   203→            fill="none" stroke={stroke} strokeWidth={2}
   204→          />
   205→          {type === 'xnor' && bubble(x + w - 6)}
   206→        </>
   207→      );
   208→    }
   209→    case 'not': {
   210→      const tw = w - 16;
   211→      return (
   212→        <>
   213→          <path
   214→            d={`M ${x},${y} L ${x + tw},${y + h / 2} L ${x},${y + h} Z`}
   215→            fill={fill} stroke={stroke} strokeWidth={2}
   216→          />
   217→          {bubble(x + w - 6)}
   218→        </>
   219→      );
   220→    }
   221→    default:
   222→      return null;
   223→  }
   224→}

   229→
   230→export function LogicCircuitCard() {
   231→  const activeSegment = useSimulator((s) => s.activeSegment);
   232→  const setActiveSegment = useSimulator((s) => s.setActiveSegment);
   233→  const expressionForm = useSimulator((s) => s.expressionForm);
   234→  const currentInput = useSimulator((s) => s.currentInput);
   235→  const inputBits = useSimulator((s) => s.inputBits);
   236→  const toggleBit = useSimulator((s) => s.toggleBit);
   237→  const explain = useSimulator((s) => s.explainMode);
   238→  const engines = useEngines();
   239→
   240→  const form = expressionForm === 'pos' || expressionForm === 'cpos' ? 'pos' : 'sop';
   241→  const engine = engines[activeSegment];
   242→  const terms = form === 'sop' ? engine.activeSopTerms : engine.activePosTerms;
   243→  const varNames = useMemo(
   244→    () => (inputBits === 5 ? ['E', 'D', 'C', 'B', 'A'] : ['D', 'C', 'B', 'A']),
   245→    [inputBits],
   246→  );
   247→
   248→  const circuit = useMemo(() => buildCircuit(terms, form, varNames), [terms, form, varNames]);
   249→
   250→  const [zoom, setZoom] = useState(1);
   251→  const [pan, setPan] = useState({ x: 0, y: 0 });
   252→  const [drag, setDrag] = useState<{
   253→    kind: 'pan' | 'gate';
   254→    nodeId?: string;
   255→    startX: number;
   256→    startY: number;
   257→    startClientX: number;
   258→    startClientY: number;
   259→    origX: number;
   260→    origY: number;
   261→  } | null>(null);
   262→  const [overrides, setOverrides] = useState<Record<string, { x: number; y: number }>>({});
   263→  const [selected, setSelected] = useState<string | null>(null);
   264→  const [moved, setMoved] = useState(false);
   265→  const svgRef = useRef<SVGSVGElement>(null);
   266→
   267→  const pos = useCallback(
   268→    (n: CNode) => overrides[n.id] ?? { x: n.x, y: n.y },
   269→    [overrides],
   270→  );
   271→
   272→  /* live evaluation */
   273→  const values = useMemo(() => {
   274→    const val = new Map<string, 0 | 1>();
   275→    const bitVals = varNames.map((_, i) => ((currentInput >> (inputBits - 1 - i)) & 1) as 0 | 1);
   276→    const evalNode = (n: CNode): 0 | 1 => {
   277→      const cached = val.get(n.id);
   278→      if (cached !== undefined) return cached;
   279→      let v: 0 | 1 = 0;
   280→      switch (n.type) {
   281→        case 'input':
   282→          v = bitVals[varNames.indexOf(n.varName!)] ?? 0;
   283→          break;
   284→        case 'const':
   285→          v = n.constValue ?? 0;
   286→          break;
   287→        case 'not':
   288→          v = evalNode(circuit.byId.get(n.operandIds[0])!) === 1 ? 0 : 1;
   289→          break;
   290→        case 'and':
   291→          v = n.operandIds.every((id) => evalNode(circuit.byId.get(id)!) === 1) ? 1 : 0;
   292→          break;
   293→        case 'or':
   294→          v = n.operandIds.some((id) => evalNode(circuit.byId.get(id)!) === 1) ? 1 : 0;
   295→          break;
   296→        case 'xor':
   297→          v = n.operandIds.reduce((acc, id) => acc ^ evalNode(circuit.byId.get(id)!), 0) as 0 | 1;
   298→          break;
   299→        case 'nand':
   300→          v = n.operandIds.every((id) => evalNode(circuit.byId.get(id)!) === 1) ? 0 : 1;
   301→          break;
   302→        case 'nor':
   303→          v = n.operandIds.some((id) => evalNode(circuit.byId.get(id)!) === 1) ? 0 : 1;
   304→          break;
   305→        case 'xnor':
   306→          v = (n.operandIds.reduce((acc, id) => acc ^ evalNode(circuit.byId.get(id)!), 0) as 0 | 1) === 1 ? 0 : 1;
   307→          break;
   308→        case 'out':
   309→          v = evalNode(circuit.byId.get(n.operandIds[0])!);
   310→          break;
   311→      }
   312→      val.set(n.id, v);
   313→      return v;
   314→    };
   315→    for (const n of circuit.nodes) evalNode(n);
   316→    return val;
   317→  }, [circuit, currentInput, inputBits, varNames]);
   318→
   319→  /* non-passive wheel zoom */
   320→  useEffect(() => {
   321→    const el = svgRef.current;
   322→    if (!el) return;
   323→    const handler = (e: WheelEvent) => {
   324→      e.preventDefault();
   325→      const factor = e.deltaY < 0 ? 1.12 : 1 / 1.12;
   326→      setZoom((z) => Math.min(2.5, Math.max(0.35, z * factor)));
   327→    };
   328→    el.addEventListener('wheel', handler, { passive: false });
   329→    return () => el.removeEventListener('wheel', handler);
   330→  }, []);
   331→
   332→  /* pointer interactions */
   333→  const toGraph = (clientX: number, clientY: number) => {
   334→    const rect = svgRef.current?.getBoundingClientRect();
   335→    if (!rect) return { x: 0, y: 0 };
   336→    // uniform viewBox scale + letterbox centering (preserveAspectRatio meet)
   337→    const scale = Math.min(circuit.width / rect.width, circuit.height / rect.height);
   338→    const offX = (rect.width - circuit.width / scale) / 2;
   339→    const offY = (rect.height - circuit.height / scale) / 2;
   340→    const vx = (clientX - rect.left - offX) * scale;
   341→    const vy = (clientY - rect.top - offY) * scale;
   342→    return { x: (vx - pan.x) / zoom, y: (vy - pan.y) / zoom };
   343→  };
   344→
   345→  const onPointerMove = (e: React.PointerEvent) => {
   346→    if (!drag) return;
   347→    const g = toGraph(e.clientX, e.clientY);
   348→    if (Math.abs(g.x - drag.startX) + Math.abs(g.y - drag.startY) > 4) setMoved(true);
   349→    if (drag.kind === 'pan') {
   350→      setPan({ x: drag.origX + (e.clientX - drag.startClientX), y: drag.origY + (e.clientY - drag.startClientY) });
   351→    } else if (drag.nodeId) {
   352→      setOverrides((o) => ({
   353→        ...o,
   354→        [drag.nodeId!]: {
   355→          x: Math.max(4, drag.origX + (g.x - drag.startX)),
   356→          y: Math.max(4, drag.origY + (g.y - drag.startY)),
   357→        },
   358→      }));
   359→    }
   360→  };
   361→
   362→  const selectedNode = selected ? circuit.byId.get(selected) : null;
   363→  const gateCount = circuit.nodes.filter((n) => ['and', 'or', 'not', 'xor', 'nand', 'nor', 'xnor'].includes(n.type)).length;
   364→  const varIndex = (name: string) => varNames.indexOf(name);
   365→
   366→  return (
   367→    <Card id="circuit-card">
   368→      <CardHeader className="pb-3">
   369→        <div className="flex flex-wrap items-center justify-between gap-2">
   370→          <CardTitle className="flex items-center gap-2 text-sm font-semibold tracking-wide">
   371→            <CircuitBoard className="h-4 w-4 text-muted-foreground" aria-hidden />
   372→            LOGIC CIRCUIT
   373→          </CardTitle>
   374→          <div className="flex flex-wrap items-center gap-1.5">
   375→            <Badge variant="secondary" className="font-mono text-[10px]">
   376→              {form.toUpperCase()} · {gateCount} gates
   377→            </Badge>
   378→            <Button size="sm" variant="outline" className="h-7 px-2" onClick={() => { setZoom((z) => Math.min(2.5, z * 1.2)); }} aria-label="Zoom in">
   379→              <ZoomIn className="h-3.5 w-3.5" aria-hidden />
   380→            </Button>
   381→            <Button size="sm" variant="outline" className="h-7 px-2" onClick={() => { setZoom((z) => Math.max(0.35, z / 1.2)); }} aria-label="Zoom out">
   382→              <ZoomOut className="h-3.5 w-3.5" aria-hidden />
   383→            </Button>
   384→            <Button size="sm" variant="outline" className="h-7 px-2" onClick={() => { setZoom(1); setPan({ x: 0, y: 0 }); }} aria-label="Fit view">
   385→              <Maximize className="h-3.5 w-3.5" aria-hidden />
   386→            </Button>
   387→            <Button size="sm" variant="outline" className="h-7 px-2" onClick={() => { setOverrides({}); setPan({ x: 0, y: 0 }); setZoom(1); }} aria-label="Reset gate layout">
   388→              <RotateCcw className="h-3.5 w-3.5" aria-hidden />
   389→            </Button>
   390→          </div>
   391→        </div>
   392→      </CardHeader>
   393→      <CardContent className="space-y-3">
   394→        {/* segment selector */}
   395→        <div className="flex flex-wrap gap-1.5" role="tablist" aria-label="Circuit segment selector">
   396→          {SEGMENT_NAMES.map((seg) => (
   397→            <button
   398→              key={seg}
   399→              type="button"
   400→              role="tab"
   401→              aria-selected={seg === activeSegment}
   402→              onClick={() => { setSelected(null); setActiveSegment(seg); }}
   403→              className={cn(
   404→                'h-8 min-w-9 rounded-md border px-2 font-mono text-xs font-semibold transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring',
   405→                seg === activeSegment
   406→                  ? 'border-primary bg-primary text-primary-foreground'
   407→                  : 'border-border bg-muted/40 text-muted-foreground hover:border-muted-foreground/40',
   408→              )}
   409→            >
   410→              {seg}
   411→            </button>
   412→          ))}
   413→        </div>
   414→
   415→        {/* SVG canvas */}
   416→        <div className="overflow-hidden rounded-lg border bg-[radial-gradient(circle,rgba(128,128,128,0.18)_1px,transparent_1px)] [background-size:22px_22px]">
   417→          <svg
   418→            ref={svgRef}
   419→            viewBox={`0 0 ${circuit.width} ${circuit.height}`}
   420→            className="block h-[400px] w-full touch-none"
   421→            role="img"
   422→            aria-label={`Logic circuit for segment ${activeSegment}, ${form.toUpperCase()} form`}
   423→            onWheel={undefined}
   424→            onPointerDown={(e) => {
   425→              if ((e.target as Element).tagName === 'svg' || (e.target as Element).classList.contains('circuit-bg')) {
   426→                setDrag({ kind: 'pan', startX: e.clientX, startY: e.clientY, startClientX: e.clientX, startClientY: e.clientY, origX: pan.x, origY: pan.y });
   427→                setMoved(false);
   428→              }
   429→            }}
   430→            onPointerMove={onPointerMove}
   431→            onPointerUp={() => {
   432→              if (drag && !moved && drag.nodeId) {
   433→                if (drag.nodeId.startsWith('in-')) {
   434→                  const idx = varIndex(drag.nodeId.slice(3));
   435→                  if (idx >= 0) toggleBit(idx);
   436→                } else {
   437→                  const id = drag.nodeId;
   438→                  setSelected((s) => (s === id ? null : id));
   439→                }
   440→              }
   441→              setDrag(null);
   442→            }}
   443→            onPointerLeave={() => setDrag(null)}
   444→          >
   445→            <rect className="circuit-bg" x={0} y={0} width={circuit.width} height={circuit.height} fill="transparent" />
   446→            <g transform={`translate(${pan.x},${pan.y}) scale(${zoom})`}>
   447→              {/* wires */}
   448→              {circuit.wires.map((w) => {
   449→                const from = circuit.byId.get(w.fromId)!;
   450→                const to = circuit.byId.get(w.toId)!;
   451→                const fp = pos(from);
   452→                const tp = pos(to);
   453→                const sx = fp.x + from.w;
   454→                const sy = fp.y + from.h / 2;
   455→                const toH = to.type === 'out' ? to.h : to.h;
   456→                const py = tp.y + ((w.pin + 1) * toH) / (to.operandIds.length + 1);
   457→                const ex = tp.x;
   458→                const ey = py;
   459→                const mx = (sx + ex) / 2;
   460→                const v = values.get(w.fromId) ?? 0;
   461→                return (
   462→                  <path
   463→                    key={w.id}
   464→                    d={`M ${sx},${sy} C ${mx},${sy} ${mx},${ey} ${ex},${ey}`}
   465→                    fill="none"
   466→                    stroke={v === 1 ? '#10b981' : 'rgba(128,128,128,0.45)'}
   467→                    strokeWidth={v === 1 ? 2.6 : 1.6}
   468→                    strokeDasharray={v === 1 ? '7 4' : undefined}
   469→                    className={v === 1 ? 'wire-active' : undefined}
   470→                    data-value={v}
   471→                  />
   472→                );
   473→              })}

   492→                      }}
   493→                    >
   494→                      <rect

   500→                      <text x={n.w / 2} y={20} textAnchor="middle" className="fill-foreground font-mono text-[13px] font-bold">

   507→                        {v}
   508→                      </text>
   509→                      <title>{`Input ${n.varName} = ${v}. Drag to move; click to toggle.`}</title>
   510→                    </g>
   511→                  );
   512→                }
   513→                if (n.type === 'out') {
   514→                  return (
   515→                    <g key={n.id} transform={`translate(${p.x},${p.y})`}>
   516→                      <rect
   517→                        width={n.w} height={n.h} rx={10}
   518→                        fill={active ? 'rgba(16,185,129,0.12)' : 'rgba(128,128,128,0.06)'}
   519→                        stroke={active ? '#10b981' : 'rgba(128,128,128,0.5)'}
   520→                        strokeWidth={isSel ? 2.5 : 1.5}
   521→                      />
   522→                      <text x={n.w / 2} y={22} textAnchor="middle" className="fill-muted-foreground font-mono text-[9px] tracking-widest">
   523→                        SEGMENT
   524→                      </text>
   525→                      <text x={n.w / 2} y={40} textAnchor="middle" className="fill-foreground font-mono text-[15px] font-bold">
   526→                        {activeSegment}
   527→                      </text>
   528→                      <circle
   529→                        cx={n.w / 2} cy={54} r={7}
   530→                        fill={active ? '#10b981' : 'rgba(128,128,128,0.3)'}
   531→                        stroke={active ? '#10b981' : 'rgba(128,128,128,0.5)'}
   532→                      />
   533→                      <text x={n.w / 2} y={57.5} textAnchor="middle" className="fill-background font-mono text-[8px] font-bold">
   534→                        {v}
   535→                      </text>
   536→                      <title>{`Segment ${activeSegment} output = ${v}`}</title>
   537→                    </g>
   538→                  );
   539→                }
   540→                if (n.type === 'const') {
   541→                  return (
   542→                    <g key={n.id} transform={`translate(${p.x},${p.y})`}>
   543→                      <rect width={n.w} height={n.h} rx={18} fill="rgba(128,128,128,0.08)" stroke="rgba(128,128,128,0.5)" strokeWidth={1.5} />
   544→                      <text x={n.w / 2} y={n.h / 2 + 5} textAnchor="middle" className="fill-foreground font-mono text-[13px] font-bold">
   545→                        {n.label}
   546→                      </text>
   547→                      <title>{`Constant ${n.label}`}</title>
   548→                    </g>
   549→                  );
   550→                }
   551→                // logic gates
   552→                return (
   553→                  <g
   554→                    key={n.id}
   555→                    transform={`translate(${p.x},${p.y})`}
   556→                    className="cursor-grab active:cursor-grabbing"
   557→                    onPointerDown={(e) => {
   558→                      e.stopPropagation();
   559→                      const g = toGraph(e.clientX, e.clientY);
   560→                      setDrag({ kind: 'gate', nodeId: n.id, startX: g.x, startY: g.y, startClientX: e.clientX, startClientY: e.clientY, origX: p.x, origY: p.y });
   561→                      setMoved(false);
   562→                    }}
   563→                  >
   564→                    {gateSymbol(n.type, 0, 0, n.w, n.h, active)}

   567→                      return <circle key={i} cx={0} cy={py} r={2.5} fill={active ? '#10b981' : 'rgba(128,128,128,0.7)'} />;
   568→                    })}
   569→                    <circle
   570→                      cx={n.w + (['nand', 'nor', 'xnor', 'not'].includes(n.type) ? 12 : 2)}
   571→                      cy={n.h / 2} r={5}
   572→                      fill={active ? '#10b981' : 'rgba(128,128,128,0.25)'}
   573→                      stroke={active ? '#10b981' : 'rgba(128,128,128,0.5)'}
   574→                    />
   575→                    <text
   576→                      x={n.w / 2 - 4} y={n.h / 2 + 4} textAnchor="middle"
   577→                      className={cn('font-mono text-[10px] font-bold pointer-events-none', active ? 'fill-emerald-500' : 'fill-muted-foreground')}

   579→                      {n.type.toUpperCase()}
   580→                    </text>
   581→                    {isSel && (
   582→                      <rect x={-6} y={-6} width={n.w + 18} height={n.h + 12} rx={8} fill="none" stroke="#f59e0b" strokeWidth={2} strokeDasharray="5 3" />
   583→                    )}
   584→                    <title>{`${n.type.toUpperCase()} gate — click to inspect, drag to move`}</title>
   585→                  </g>
   586→                );
   587→              })}
   588→            </g>
   589→          </svg>
   590→        </div>
   591→
   592→        {/* legend + hint */}
   593→        <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-[11px] text-muted-foreground">
   594→          <span className="flex items-center gap-1.5">
   595→            <span className="h-0.5 w-6 rounded bg-emerald-500" aria-hidden />
   596→            logic&nbsp;1 (active path)
   597→          </span>
   598→          <span className="flex items-center gap-1.5">
   599→            <span className="h-0.5 w-6 rounded bg-muted-foreground/40" aria-hidden />
   600→            logic&nbsp;0
   601→          </span>
   602→          <span className="flex items-center gap-1.5">
   603→            <MousePointerClick className="h-3.5 w-3.5" aria-hidden />
   604→            click inputs to toggle · click gates to inspect · drag to rearrange · wheel to zoom
   605→          </span>
   606→        </div>
   607→
   608→        {/* inspect panel */}
   609→        {selectedNode && (
   610→          <div className="rounded-lg border border-amber-500/40 bg-amber-500/5 p-3">
   611→            <div className="flex flex-wrap items-center gap-2 text-sm">
   612→              <Lightbulb className="h-4 w-4 text-amber-500" aria-hidden />
   613→              <span className="font-mono font-semibold">{selectedNode.type.toUpperCase()}</span>
   614→              <span className="text-muted-foreground">
   615→                {selectedNode.type === 'input'
   616→                  ? `input variable ${selectedNode.varName}`
   617→                  : selectedNode.type === 'out'
   618→                    ? `segment output driver`

   621→              <Badge className={cn(
   622→                'font-mono',
   623→                (values.get(selectedNode.id) ?? 0) === 1
   624→                  ? 'bg-emerald-500/15 text-emerald-600 dark:text-emerald-400'

   627→                out = {values.get(selectedNode.id) ?? 0}
   628→              </Badge>
   629→            </div>
   630→            <p className="mt-1 text-xs text-muted-foreground">
   631→              {GATE_INFO[selectedNode.type.toUpperCase()]?.desc ??
   632→                (selectedNode.type === 'out'
   633→                  ? `Drives segment ${activeSegment} for every input combination.`
   634→                  : `Current value with input m${currentInput}.`)}
   635→              {GATE_INFO[selectedNode.type.toUpperCase()] && (
   636→                <span className="ml-1 font-mono">{GATE_INFO[selectedNode.type.toUpperCase()].expr}</span>
   637→              )}
   638→            </p>
   639→            {selectedNode.operandIds.length > 0 && (
   640→              <div className="mt-2 flex flex-wrap gap-1.5">
   641→                {selectedNode.operandIds.map((id, i) => {
   642→                  const src = circuit.byId.get(id)!;
   643→                  const sv = values.get(id) ?? 0;
   644→                  return (
   645→                    <span
   646→                      key={id}
   647→                      className={cn(
   648→                        'rounded border px-1.5 py-0.5 font-mono text-[10px]',
   649→                        sv === 1
   650→                          ? 'border-emerald-500/40 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400'
   651→                          : 'border-border bg-muted/40 text-muted-foreground',
   652→                      )}
   653→                    >
   654→                      in{i + 1}: {src.varName ?? src.type} = {sv}

   657→                })}
   658→              </div>
   659→            )}
   660→            <p className="mt-2 text-[11px] text-muted-foreground">
   661→              Implemented from the simplified {form.toUpperCase()} expression:{' '}
   662→              {terms.length === 0
   663→                ? form === 'sop' ? 'constant 0' : 'constant 1'
   664→                : terms.map((t, i) => (
   665→                    <span key={i} className="mr-1 inline-flex items-center">
   666→                      {i > 0 && <span className="mr-1 text-muted-foreground">{form === 'sop' ? '+' : '·'}</span>}
   667→                      <TermView literals={t.literals} op={form === 'sop' ? '·' : '+'} />
   668→                    </span>
   669→                  ))}
   670→            </p>
   671→          </div>
   672→        )}
   673→
   674→        {/* gate reference */}
   675→        <details className="rounded-lg border bg-muted/20 p-3">
   676→          <summary className="cursor-pointer text-xs font-semibold uppercase tracking-widest text-muted-foreground">
   677→            Gate reference (AND · OR · NOT · NAND · NOR · XOR · XNOR)
   678→          </summary>
   679→          <div className="mt-3 grid grid-cols-2 gap-3 sm:grid-cols-4 lg:grid-cols-7">

   683→                  {gateSymbol(t, 6, 4, 62, 36, false)}
   684→                </svg>
   685→                <span className="font-mono text-[10px] font-semibold">{t.toUpperCase()}</span>
   686→                <span className="text-center font-mono text-[9px] leading-tight text-muted-foreground">
   687→                  {GATE_INFO[t.toUpperCase()].expr}
   688→                </span>
   689→              </div>
   690→            ))}
   691→          </div>
   692→          <Separator className="my-3" />
   693→          <div className="grid gap-2 text-[11px] text-muted-foreground sm:grid-cols-2">
   694→            {(['and', 'or', 'not', 'xor'] as GateType[]).map((t) => (
   695→              <div key={t} className="flex items-center gap-2">
   696→                <span className="w-10 font-mono font-semibold text-foreground/80">{t.toUpperCase()}</span>
   697→                <span>{GATE_INFO[t.toUpperCase()].desc}</span>
   698→              </div>
   699→            ))}
   700→          </div>
   701→        </details>
   702→
   703→        {explain && (

   705→            <p>
   706→              The circuit is drawn directly from the minimized {form.toUpperCase()} expression with{' '}

   711→              gates generate the complemented rails.

   714→              With the current input <span className="font-mono">m{currentInput}</span>, the highlighted

   717→              <span className="font-mono">{values.get('out') ?? 0}</span> for this input, matching the
   718→              truth table.
   719→            </p>
   720→            <p>
   721→              Input rails are clickable (they toggle the simulator input), gates are draggable, and
   722→              clicking a gate shows its live inputs and output. The number of levels equals the gate
   723→              depth: rails → NOT → {form === 'sop' ? 'AND' : 'OR'} → {form === 'sop' ? 'OR' : 'AND'} → segment.
   724→            </p>
   725→          </ExplainBlock>
   726→        )}
   727→      </CardContent>
