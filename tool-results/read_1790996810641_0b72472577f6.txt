     1→/**
     2→ * Logic-gate circuit visualizer, generated from the (minimized OR canonical)
     3→ * Boolean expression of the selected segment — or from all eight segments at
     4→ * once in the full-decoder view.
     5→ *
     6→ * This module is the composition root + view-state owner: view mode
     7→ * (segment / full decoder), zoom/pan/gate-drag overrides/selection (all of
     8→ * which survive tab switches while this card stays mounted), fit-to-content,
     9→ * fullscreen, pointer + keyboard interaction math, the input auto-sweep
    10→ * player, the level trace (propagation step-through), live signal evaluation
    11→ * and the circuit image exports. The SVG canvas, gate inspector and gate
    12→ * reference live in sibling files; the pure netlist builders/evaluators live
    13→ * in @/logic/circuitNetlist and @/logic/decoderCircuit.
    14→ */

    51→const ZOOM_MIN = 0.1;
    52→const ZOOM_MAX = 2.5;
    53→const GATE_TYPES = ['and', 'or', 'not', 'xor', 'nand', 'nor', 'xnor'] as const;
    54→
    55→const FORM_CHIPS: { form: ExpressionForm; label: string; title: string }[] = [
    56→  { form: 'sop', label: 'SOP', title: 'Simplified SOP (minimized product terms)' },
    57→  { form: 'pos', label: 'POS', title: 'Simplified POS (minimized sum terms)' },
    58→  { form: 'csop', label: 'Σm', title: 'Canonical SOP — one full minterm per 1-row (unminimized)' },
    59→  { form: 'cpos', label: 'ΠM', title: 'Canonical POS — one full maxterm per 0-row (unminimized)' },
    60→];
    61→
    62→const SWEEP_SPEEDS = [
    63→  { ms: 500, label: '1×' },
    64→  { ms: 250, label: '2×' },

    67→
    68→export function LogicCircuitCard() {
    69→  const activeSegment = useSimulator((s) => s.activeSegment);
    70→  const setActiveSegment = useSimulator((s) => s.setActiveSegment);
    71→  const expressionForm = useSimulator((s) => s.expressionForm);
    72→  const setExpressionForm = useSimulator((s) => s.setExpressionForm);
    73→  const currentInput = useSimulator((s) => s.currentInput);
    74→  const inputBits = useSimulator((s) => s.inputBits);
    75→  const toggleBit = useSimulator((s) => s.toggleBit);
    76→  const incrementInput = useSimulator((s) => s.incrementInput);
    77→  const explain = useSimulator((s) => s.explainMode);
    78→  const displaySettings = useSimulator((s) => s.displaySettings);
    79→  const projectName = useSimulator((s) => s.projectName);
    80→  const engines = useEngines();
    81→
    82→  const form = expressionForm === 'pos' || expressionForm === 'cpos' ? 'pos' : 'sop';
    83→  const isCanonical = expressionForm === 'csop' || expressionForm === 'cpos';
    84→  const varNames = useSimulator((s) => s.variableNames);
    85→
    86→  /* ---------------------------------------------------------------- */
    87→  /* Terms per segment — canonical forms build REAL unminimized        */
    88→  /* netlists (minterm/maxterm terms), so canonical vs minimized is    */
    89→  /* visible in the gate count and layout, not just the label.         */
    90→  /* ---------------------------------------------------------------- */
    91→  const allTerms = useMemo(() => {
    92→    const out = {} as Record<SegmentName, Term[]>;
    93→    for (const seg of SEGMENT_NAMES) {
    94→      const e = engines[seg];
    95→      if (!isCanonical) {
    96→        out[seg] = form === 'sop' ? e.activeSopTerms : e.activePosTerms;
    97→      } else {
    98→        out[seg] = form === 'sop'
    99→          ? e.ones.map((m) => mintermToTerm(m, inputBits, varNames))
   100→          : e.zeros.map((m) => maxtermToTerm(m, inputBits, varNames));
   101→      }
   102→    }
   103→    return out;
   104→  }, [engines, form, isCanonical, inputBits, varNames]);
   105→
   106→  /* ---------------------- view state (local) ---------------------- */
   107→  const [mode, setMode] = useState<'segment' | 'decoder'>('segment');
   108→  const [expanded, setExpanded] = useState(false);
   109→  const [zoom, setZoom] = useState(1);
   110→  const [pan, setPan] = useState({ x: 0, y: 0 });
   111→  const [drag, setDrag] = useState<{
   112→    kind: 'pan' | 'gate';
   113→    nodeId?: string;
   114→    startX: number;
   115→    startY: number;
   116→    startClientX: number;
   117→    startClientY: number;
   118→    origX: number;
   119→    origY: number;
   120→  } | null>(null);
   121→  const [overrides, setOverrides] = useState<Record<string, { x: number; y: number }>>({});
   122→  const [selected, setSelected] = useState<string | null>(null);
   123→  const [moved, setMoved] = useState(false);
   124→  const [playing, setPlaying] = useState(false);
   125→  const [sweepIdx, setSweepIdx] = useState(0);
   126→  const [trace, setTrace] = useState(false);
   127→  const [revealLevel, setRevealLevel] = useState(0);
   128→  const [exporting, setExporting] = useState<string | null>(null);
   129→  const svgRef = useRef<SVGSVGElement>(null);
   130→
   131→  const pos = useCallback(
   132→    (n: CNode) => overrides[n.id] ?? { x: n.x, y: n.y },
   133→    [overrides],
   134→  );
   135→
   136→  /* --------------------------- circuits --------------------------- */
   137→  const decoder = useMemo(
   138→    () => (mode === 'decoder' ? buildDecoderCircuit(allTerms, form, varNames) : null),
   139→    [mode, allTerms, form, varNames],
   140→  );
   141→  const circuit = useMemo(
   142→    () => (mode === 'decoder'
   143→      ? decoder!.circuit
   144→      : buildCircuit(allTerms[activeSegment], form, varNames)),
   145→    [mode, decoder, allTerms, activeSegment, form, varNames],
   146→  );
   147→
   148→  /* live evaluation */
   149→  const values = useMemo(
   150→    () => evaluateCircuit(circuit, currentInput, inputBits, varNames),
   151→    [circuit, currentInput, inputBits, varNames],
   152→  );
   153→
   154→  /** Propagation depth per node (inputs = 0) — powers the level trace. */
   155→  const levels = useMemo(() => computeLevels(circuit), [circuit]);
   156→  const maxLevel = useMemo(() => Math.max(0, ...levels.values()), [levels]);
   157→
   158→  /**
   159→   * Live logic levels of ALL eight segments for the current input: every
   160→   * segment's minimized circuit ({form}) is evaluated, so the glyph shows
   161→   * exactly what the gate network drives right now. Don't-care cells resolve
   162→   * to the cover's actual choice — the physical truth the circuit delivers.
   163→   */
   164→  const liveOutputs = useMemo(() => {
   165→    const out: SegmentPatternX = {} as SegmentPatternX;
   166→    for (const seg of SEGMENT_NAMES) {
   167→      const segCircuit = buildCircuit(allTerms[seg], form, varNames);
   168→      out[seg] = evaluateCircuit(segCircuit, currentInput, inputBits, varNames).get('out') ?? 0;
   169→    }
   170→    return out;
   171→  }, [allTerms, form, varNames, currentInput, inputBits]);
   172→
   173→  /** Logic-level view of the display: user colors/brightness, but polarity
   174→   *  forced to common-cathode — this panel shows gate outputs, not the
   175→   *  physically-driven LED (that is the main display card's job). */
   176→  const liveSettings = useMemo(
   177→    () => ({ ...displaySettings, commonAnode: false as const, reverseLogic: false, showDP: true }),
   178→    [displaySettings],
   179→  );
   180→
   181→  /* Selected node's logic cone: transitive fan-in + fan-out. */
   182→  const cone = useMemo(() => {
   183→    if (!selected || !circuit.byId.has(selected)) return null;
   184→    const nodeIds = new Set<string>([selected]);
   185→    const stack = [selected];
   186→    while (stack.length) {
   187→      const n = circuit.byId.get(stack.pop()!)!;
   188→      for (const id of n.operandIds) {
   189→        if (!nodeIds.has(id)) { nodeIds.add(id); stack.push(id); }
   190→      }
   191→    }
   192→    const rev = new Map<string, string[]>();
   193→    for (const n of circuit.nodes) {
   194→      for (const id of n.operandIds) rev.set(id, [...(rev.get(id) ?? []), n.id]);
   195→    }
   196→    stack.push(selected);
   197→    while (stack.length) {
   198→      const id = stack.pop()!;
   199→      for (const d of rev.get(id) ?? []) {
   200→        if (!nodeIds.has(d)) { nodeIds.add(d); stack.push(d); }
   201→      }
   202→    }
   203→    const wireIds = new Set(
   204→      circuit.wires.filter((w) => nodeIds.has(w.fromId) && nodeIds.has(w.toId)).map((w) => w.id),
   205→    );
   206→    return { nodeIds, wireIds };
   207→  }, [selected, circuit]);
   208→
   209→  /**
   210→   * K-map cover-group accents: every term gate in the (minimized) circuit
   211→   * gets the color of its group in the K-map tab — the visible link between
   212→   * the two views. Canonical forms have no per-term groups, so no accents.
   213→   */
   214→  const nodeAccents = useMemo(() => {
   215→    const map = new Map<string, NodeAccent>();
   216→    if (isCanonical) return map;
   217→    for (const seg of SEGMENT_NAMES) {
   218→      if (mode === 'segment' && seg !== activeSegment) continue;
   219→      const groups = form === 'sop' ? engines[seg].sopGroups : engines[seg].posGroups;
   220→      allTerms[seg].forEach((t, ti) => {
   221→        const key = termGroupKey(t);
   222→        const g = groups.find((gr) => termGroupKey(gr.term) === key);
   223→        if (!g) return;
   224→        const id = mode === 'segment' ? `term-${ti}` : `t-${seg}-${ti}`;
   225→        map.set(id, {
   226→          color: groupColor(g.colorIndex),
   227→          title: `K-map group ${key.replace(/\*/g, "'")} · ${g.size} cells${g.isEssential ? ' · essential' : ''}`,
   228→        });
   229→      });
   230→    }
   231→    return map;
   232→  }, [engines, form, isCanonical, mode, activeSegment, allTerms]);
   233→
   234→  /* ------------------------- fit to content ----------------------- */
   235→  const fitView = useCallback(() => {
   236→    const el = svgRef.current;
   237→    if (!el) return;
   238→    const rect = el.getBoundingClientRect();
   239→    if (rect.width < 10 || rect.height < 10) return;
   240→    let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
   241→    for (const n of circuit.nodes) {
   242→      const p = overrides[n.id] ?? { x: n.x, y: n.y };
   243→      minX = Math.min(minX, p.x);
   244→      minY = Math.min(minY, p.y);
   245→      maxX = Math.max(maxX, p.x + n.w);
   246→      maxY = Math.max(maxY, p.y + n.h);
   247→    }
   248→    if (!Number.isFinite(minX)) return;
   249→    const bw = Math.max(1, maxX - minX);
   250→    const bh = Math.max(1, maxY - minY);
   251→    // viewBox → screen mapping (uniform meet) — must mirror toGraph()
   252→    const vbScale = Math.min(circuit.width / rect.width, circuit.height / rect.height);
   253→    const k = 1 / vbScale;
   254→    const offX = (rect.width - circuit.width / vbScale) / 2;
   255→    const offY = (rect.height - circuit.height / vbScale) / 2;
   256→    const pad = 24;
   257→    const z = Math.min(
   258→      ZOOM_MAX,
   259→      Math.max(ZOOM_MIN, Math.min((rect.width - 2 * pad) / (bw * k), (rect.height - 2 * pad) / (bh * k))),
   260→    );
   261→    setZoom(z);
   262→    setPan({ x: (pad - offX) / k - minX * z, y: (pad - offY) / k - minY * z });
   263→  }, [circuit, overrides]);
   264→
   265→  /** Auto-fit once per (layout × canvas size); never fights the user. */
   266→  const [lastFitKey, setLastFitKey] = useState('');
   267→  const maybeFit = useCallback(() => {
   268→    const el = svgRef.current;
   269→    if (!el) return;
   270→    const rect = el.getBoundingClientRect();
   271→    if (rect.width < 10) return;
   272→    const key = `${mode}|${activeSegment}|${expressionForm}|${inputBits}|${Math.round(rect.width)}x${Math.round(rect.height)}`;
   273→    if (key === lastFitKey) return;
   274→    setLastFitKey(key);
   275→    fitView();
   276→  }, [mode, activeSegment, expressionForm, inputBits, lastFitKey, fitView]);
   277→
   278→  useEffect(() => { maybeFit(); }, [maybeFit]);
   279→
   280→  useEffect(() => {
   281→    const el = svgRef.current;
   282→    if (!el || typeof ResizeObserver === 'undefined') return;
   283→    const ro = new ResizeObserver(() => maybeFit());
   284→    ro.observe(el);
   285→    return () => ro.disconnect();
   286→  }, [maybeFit]);
   287→
   288→  /* non-passive wheel zoom */
   289→  useEffect(() => {
   290→    const el = svgRef.current;
   291→    if (!el) return;
   292→    const handler = (e: WheelEvent) => {
   293→      e.preventDefault();
   294→      const factor = e.deltaY < 0 ? 1.12 : 1 / 1.12;
   295→      setZoom((z) => Math.min(ZOOM_MAX, Math.max(ZOOM_MIN, z * factor)));
   296→    };
   297→    el.addEventListener('wheel', handler, { passive: false });
   298→    return () => el.removeEventListener('wheel', handler);
   299→  }, []);
   300→
   301→  /* fullscreen: Esc to leave + body scroll lock */
   302→  useEffect(() => {
   303→    if (!expanded) return;
   304→    const onKey = (e: KeyboardEvent) => { if (e.key === 'Escape') setExpanded(false); };
   305→    window.addEventListener('keydown', onKey);
   306→    const prev = document.body.style.overflow;
   307→    document.body.style.overflow = 'hidden';
   308→    return () => {
   309→      window.removeEventListener('keydown', onKey);
   310→      document.body.style.overflow = prev;
   311→    };
   312→  }, [expanded]);
   313→
   314→  /* input auto-sweep player */
   315→  useEffect(() => {
   316→    if (!playing) return;
   317→    const t = setInterval(() => incrementInput(), SWEEP_SPEEDS[sweepIdx].ms);
   318→    return () => clearInterval(t);
   319→  }, [playing, sweepIdx, incrementInput]);
   320→
   321→  /* level trace resets on layout change */
   322→  useEffect(() => { setRevealLevel(0); }, [mode, activeSegment, expressionForm, inputBits]);
   323→
   324→  /* ---------------------- pointer interactions --------------------- */
   325→  const toGraph = (clientX: number, clientY: number) => {
   326→    const rect = svgRef.current?.getBoundingClientRect();
   327→    if (!rect) return { x: 0, y: 0 };
   328→    // uniform viewBox scale + letterbox centering (preserveAspectRatio meet)
   329→    const scale = Math.min(circuit.width / rect.width, circuit.height / rect.height);
   330→    const offX = (rect.width - circuit.width / scale) / 2;
   331→    const offY = (rect.height - circuit.height / scale) / 2;
   332→    const vx = (clientX - rect.left - offX) * scale;
   333→    const vy = (clientY - rect.top - offY) * scale;
   334→    return { x: (vx - pan.x) / zoom, y: (vy - pan.y) / zoom };
   335→  };
   336→
   337→  const onPointerMove = (e: React.PointerEvent) => {
   338→    if (!drag) return;
   339→    const g = toGraph(e.clientX, e.clientY);
   340→    if (Math.abs(g.x - drag.startX) + Math.abs(g.y - drag.startY) > 4) setMoved(true);
   341→    if (drag.kind === 'pan') {
   342→      setPan({ x: drag.origX + (e.clientX - drag.startClientX), y: drag.origY + (e.clientY - drag.startClientY) });
   343→    } else if (drag.nodeId) {
   344→      setOverrides((o) => ({
   345→        ...o,
   346→        [drag.nodeId!]: {
   347→          x: Math.max(4, drag.origX + (g.x - drag.startX)),
   348→          y: Math.max(4, drag.origY + (g.y - drag.startY)),
   349→        },
   350→      }));
   351→    }
   352→  };
   353→
   354→  /** Background press → start a pan drag (only when the empty canvas was hit). */
   355→  const onCanvasPointerDown = (e: React.PointerEvent<SVGSVGElement>) => {
   356→    if ((e.target as Element).tagName === 'svg' || (e.target as Element).classList.contains('circuit-bg')) {
   357→      setDrag({ kind: 'pan', startX: e.clientX, startY: e.clientY, startClientX: e.clientX, startClientY: e.clientY, origX: pan.x, origY: pan.y });
   358→      setMoved(false);
   359→    }
   360→  };
   361→
   362→  /** Node press → start a gate drag (click without moving toggles/inspects). */
   363→  const onNodePointerDown = (e: React.PointerEvent<SVGGElement>, node: CNode, p: { x: number; y: number }) => {
   364→    e.stopPropagation();
   365→    const g = toGraph(e.clientX, e.clientY);
   366→    setDrag({ kind: 'gate', nodeId: node.id, startX: g.x, startY: g.y, startClientX: e.clientX, startClientY: e.clientY, origX: p.x, origY: p.y });
   367→    setMoved(false);
   368→  };
   369→
   370→  const activateNode = useCallback((node: CNode) => {
   371→    if (node.id.startsWith('in-')) {
   372→      const idx = varNames.indexOf(node.id.slice(3));
   373→      if (idx >= 0) toggleBit(idx);
   374→    } else {
   375→      setSelected((s) => (s === node.id ? null : node.id));
   376→    }
   377→  }, [varNames, toggleBit]);
   378→
   379→  const onPointerUp = () => {
   380→    if (drag && !moved && drag.nodeId) {
   381→      const node = circuit.byId.get(drag.nodeId);
   382→      if (node) activateNode(node);
   383→    }
   384→    setDrag(null);
   385→  };
   386→
   387→  /** Canvas-level keyboard: arrows pan, +/- zoom, 0 fit. */
   388→  const onCanvasKeyDown = (e: React.KeyboardEvent<SVGSVGElement>) => {
   389→    const step = 60;
   390→    switch (e.key) {
   391→      case 'ArrowLeft': setPan((p) => ({ ...p, x: p.x + step })); e.preventDefault(); break;
   392→      case 'ArrowRight': setPan((p) => ({ ...p, x: p.x - step })); e.preventDefault(); break;
   393→      case 'ArrowUp': setPan((p) => ({ ...p, y: p.y + step })); e.preventDefault(); break;
   394→      case 'ArrowDown': setPan((p) => ({ ...p, y: p.y - step })); e.preventDefault(); break;
   395→      case '+': case '=': setZoom((z) => Math.min(ZOOM_MAX, z * 1.2)); e.preventDefault(); break;
   396→      case '-': setZoom((z) => Math.max(ZOOM_MIN, z / 1.2)); e.preventDefault(); break;
   397→      case '0': fitView(); e.preventDefault(); break;
   398→    }
   399→  };
   400→
   401→  /* --------------------------- derived UI -------------------------- */
   402→  const selectedNode = selected ? (circuit.byId.get(selected) ?? null) : null;
   403→  const ownerSegment = selectedNode && decoder ? decoder.nodeOwner.get(selectedNode.id) : undefined;
   404→  const gateCount = useMemo(
   405→    () => circuit.nodes.filter((n) => (GATE_TYPES as readonly string[]).includes(n.type)).length,
   406→    [circuit],
   407→  );
   408→  const formLabel = isCanonical
   409→    ? (form === 'sop' ? 'CANONICAL Σm' : 'CANONICAL ΠM')
   410→    : form.toUpperCase();
   411→  const outValue = mode === 'segment'
   412→    ? (values.get('out') ?? 0)
   413→    : (values.get(`o-${activeSegment}`) ?? 0);
   414→  const reveal = trace ? Math.min(revealLevel, maxLevel) : null;
   415→
   416→  /** K-map accent info for the inspector when a term gate is selected. */
   417→  const inspectorAccent = useMemo(() => {
   418→    if (!selectedNode || !nodeAccents.has(selectedNode.id)) return null;
   419→    const seg = ownerSegment ?? activeSegment;
   420→    const groups = form === 'sop' ? engines[seg].sopGroups : engines[seg].posGroups;
   421→    const terms = allTerms[seg];
   422→    const idx = mode === 'segment'
   423→      ? Number(selectedNode.id.slice('term-'.length))
   424→      : Number(selectedNode.id.slice(selectedNode.id.indexOf('-', 2) + 1));
   425→    const t = terms[idx];
   426→    const g = t ? groups.find((gr) => termGroupKey(gr.term) === termGroupKey(t)) : undefined;
   427→    if (!g) return null;
   428→    return { color: groupColor(g.colorIndex), size: g.size, essential: g.isEssential };
   429→  }, [selectedNode, nodeAccents, ownerSegment, activeSegment, engines, form, allTerms, mode]);
   430→
   431→  /* --------------------------- image export ------------------------ */
   432→  const exportCircuit = async (kind: 'svg' | 'png') => {
   433→    const svg = svgRef.current;
   434→    if (!svg) return;
   435→    setExporting(kind);
   436→    try {
   437→      const base = `${sanitizeFilename(projectName)}-circuit-${mode === 'decoder' ? 'decoder' : activeSegment}-${expressionForm}`;
   438→      if (kind === 'svg') {
   439→        downloadBlob(buildCircuitSvgBlob(svg), `${base}.svg`);
   440→        toast({ title: 'Circuit SVG exported', description: `${mode === 'decoder' ? 'Full decoder' : `Segment ${activeSegment}`} · ${formLabel}` });
   441→      } else {
   442→        const bgRaw = getComputedStyle(svg).backgroundColor;
   443→        const background = bgRaw && bgRaw !== 'rgba(0, 0, 0, 0)' ? bgRaw : '#ffffff';
   444→        downloadBlob(await buildCircuitPngBlob(svg, background), `${base}.png`);
   445→        toast({ title: 'Circuit PNG exported', description: `${mode === 'decoder' ? 'Full decoder' : `Segment ${activeSegment}`} · ${formLabel}` });
   446→      }
   447→    } catch (err) {
   448→      toast({ title: 'Export failed', description: err instanceof Error ? err.message : 'Unexpected error', variant: 'destructive' });
