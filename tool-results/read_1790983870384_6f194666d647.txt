   175→function persist(state: SimulatorState) {
   176→  if (typeof window === 'undefined') return;
   177→  try {
   178→    const data = state.toProjectData();
   179→    window.localStorage.setItem(WORKSPACE_KEY, JSON.stringify(data));
   180→  } catch {
   181→    /* storage full / unavailable — demo continues in memory */
   182→  }
   183→}
   184→
   185→/** Write the preset library to localStorage. */
   186→function persistLibrary(library: PresetLibrary) {
   187→  if (typeof window === 'undefined') return;
   188→  try {
   189→    window.localStorage.setItem(PRESET_LIB_KEY, JSON.stringify(library));
   190→  } catch {
   191→    /* storage full — library stays in memory for this session */
   192→  }
   193→}
   194→
   195→/** Load + validate the preset library from localStorage. */
   196→function loadLibrary(): PresetLibrary {
   197→  if (typeof window === 'undefined') return { tables: [], fonts: [] };
   198→  try {
   199→    const raw = window.localStorage.getItem(PRESET_LIB_KEY);
   200→    if (!raw) return { tables: [], fonts: [] };
   201→    const parsed = JSON.parse(raw) as { tables?: unknown; fonts?: unknown };
   202→    const tables = Array.isArray(parsed.tables)
   203→      ? parsed.tables.map(validateLibraryTablePreset).filter((t): t is TablePresetDef => t !== null)
   204→      : [];
   205→    const fonts = Array.isArray(parsed.fonts)
   206→      ? parsed.fonts.map(validateLibraryFontPreset).filter((f): f is FontPresetDef => f !== null)
   207→      : [];
   208→    return { tables, fonts };
   209→  } catch {
   210→    return { tables: [], fonts: [] };
   211→  }
   212→}

   214→/** Best-effort server sync of one library preset (fire and forget). */
   215→async function apiUpsertPreset(def: TablePresetDef | FontPresetDef, kind: PresetKind): Promise<void> {
   216→  try {
   217→    const res = await fetch('/api/presets', {
   218→      method: 'POST',
   219→      headers: { 'Content-Type': 'application/json' },
   220→      body: JSON.stringify({ id: def.id, kind, name: def.name, data: def }),
   221→    });
   222→    if (res.status === 409) {
   223→      await fetch(`/api/presets/${def.id}`, {
   224→        method: 'PUT',
   225→        headers: { 'Content-Type': 'application/json' },
   226→        body: JSON.stringify({ name: def.name, data: def }),
   227→      });
   228→    }
   229→  } catch {
   230→    /* offline — localStorage remains the source of truth */
   231→  }
   232→}

   234→/** Resolve a table preset by id across built-ins and the user library. */
   235→export function resolveTablePreset(id: string | null, library: PresetLibrary): TablePresetDef | null {
   236→  if (!id) return null;
   237→  return findBuiltinTablePreset(id) ?? library.tables.find((t) => t.id === id) ?? null;
   238→}

   240→/** Resolve a font preset by id across built-ins and the user library. */
   241→export function resolveFontPreset(id: string, library: PresetLibrary): FontPresetDef | null {
   242→  const builtin = findBuiltinFont(id);
   243→  if (builtin) return builtin;
   244→  return library.fonts.find((f) => f.id === id) ?? null;
   245→}

   247→/** Active font glyph list: builtin ids select builtin fonts; otherwise customFont wins. */
   248→export function activeFontGlyphs(customFont: CustomFont | null, activeFontId = 'builtin:chart'): FontGlyph[] {
   249→  if (activeFontId === 'builtin:calculator') return CALCULATOR_GLYPHS as FontGlyph[];
   250→  if (activeFontId === 'builtin:chart') return FONT_GLYPHS;
   251→  return customFont?.glyphs ?? FONT_GLYPHS;
   252→}

   257→/** Synthesize the builtin table preset id/name implied by a display mode. */
   258→function builtinForMode(mode: DisplayMode): { id: string; name: string } | null {
   259→  if (mode === 'decimal') {
   260→    const p = findBuiltinTablePreset('builtin:bcd')!;
   261→    return { id: p.id, name: p.name };
   262→  }
   263→  if (mode === 'hex') {
   264→    const p = findBuiltinTablePreset('builtin:hex')!;
   265→    return { id: p.id, name: p.name };
   266→  }
   267→  if (mode === 'font') {
   268→    const p = findBuiltinTablePreset('builtin:font')!;
   269→    return { id: p.id, name: p.name };
   270→  }
   271→  return null;
   272→}

   274→/**
   275→ * Make customFont / activeFontId / library consistent:
   276→ * a user font id must have a library entry AND a customFont blob.
   277→ * Returns the patch to apply (empty object when already consistent).
   278→ */
   279→function reconcileFont(
   280→  customFont: CustomFont | null,
   281→  activeFontId: string,
   282→  library: PresetLibrary,
   283→): { customFont?: CustomFont | null; activeFontId?: string; library?: PresetLibrary } {
   284→  if (activeFontId === 'builtin:chart' || activeFontId === 'builtin:calculator') {
   285→    return customFont ? { customFont: null } : {};
   286→  }
   287→  // user font id (or legacy sentinel)
   288→  const entry = library.fonts.find((f) => f.id === activeFontId);
   289→  if (entry) {
   290→    if (customFont && customFont.name === entry.name
   291→      && JSON.stringify(customFont.glyphs) === JSON.stringify(entry.glyphs)) return {};
   292→    return { customFont: { name: entry.name, glyphs: entry.glyphs } };
   293→  }
   294→  // not in library — synthesize from customFont when possible
   295→  if (customFont) {
   296→    const id = activeFontId === FONT_CUSTOM_SENTINEL ? newUserId() : activeFontId;
   297→    const def: FontPresetDef = {
   298→      id,
   299→      name: customFont.name,
   300→      builtin: false,
   301→      glyphs: customFont.glyphs,
   302→      createdAt: new Date().toISOString(),
   303→      updatedAt: new Date().toISOString(),
   304→    };
   305→    return {
   306→      activeFontId: id,
   307→      library: { tables: library.tables, fonts: [...library.fonts, def] },
   308→    };
   309→  }
   310→  return { activeFontId: 'builtin:chart' };
   311→}

   468→  setInputBits: (bits) => {
   469→    if (bits === get().inputBits) return;
   470→    const fontMode = get().displayMode === 'font';
   471→    const applied = resolveTablePreset(get().appliedPresetId, get().presetLibrary);
   472→    const regen = applied && ['excess3', 'gray', 'aiken', 'allon', 'alloff'].includes(applied.generator);
   473→    let table: TruthTable;
   474→    if (fontMode) {
   475→      table = presetTable(bits, 'font', get().invalidBehavior, undefined, {
   476→        fontStart: get().fontStart,
   477→        tableStartCode: get().tableStartCode,
   478→        glyphs: activeFontGlyphs(get().customFont, get().activeFontId),
   479→      });
   480→    } else if (regen && applied) {
   481→      table = buildPresetTable(applied, bits, get().invalidBehavior, activeFontGlyphs(get().customFont, get().activeFontId));
   482→    } else {
   483→      const resized = resizeTable(get().truthTable, bits);
   484→      table = applyInvalidBehavior(resized, get().displayMode, get().invalidBehavior);
   485→    }
   486→    const max = maxvalue(bits);
   487→    set({
   488→      inputBits: bits,
   489→      truthTable: table,
   490→      currentInput: Math.min(get().currentInput, max),
   491→      tableStartCode: Math.min(get().tableStartCode, max),
   492→      variableNames: inputVarNames(bits),
   493→      undoSnapshot: null,
   494→      lastAction: `Switched to ${bits}-bit input (${inputVarNames(bits).join(' ')})`,
   495→    });
   496→    persist(get());
   497→  },
   498→
   499→  setInput: (value) => {
   500→    const max = maxvalue(get().inputBits);
   501→    const v = Math.min(Math.max(0, Math.round(value)), max);
   502→    if (v !== get().currentInput) set({ currentInput: v });
   503→  },
   504→
   505→  toggleBit: (msbIndex) => {
   506→    const { currentInput, inputBits } = get();
   507→    const bitPos = inputBits - 1 - msbIndex; // convert MSB index to weight
   508→    const next = currentInput ^ (1 << bitPos);
   509→    set({ currentInput: next });
   510→  },
   511→
   512→  incrementInput: () => {
   513→    const { currentInput, inputBits } = get();
   514→    const max = maxvalue(inputBits);
   515→    set({ currentInput: currentInput >= max ? 0 : currentInput + 1 });
   516→  },
   517→
   518→  decrementInput: () => {
   519→    const { currentInput, inputBits } = get();
   520→    const max = maxvalue(inputBits);
   521→    set({ currentInput: currentInput <= 0 ? max : currentInput - 1 });
   522→  },
   523→
   524→  setDisplayMode: (mode) => {
   525→    const prev = get().displayMode;
   526→    if (prev === mode) return;
   527→    const table = mode === 'font'
   528→      ? presetTable(get().inputBits, 'font', get().invalidBehavior, undefined, {
   529→          fontStart: get().fontStart,
   530→          tableStartCode: get().tableStartCode,

   921→  syncPresetsFromServer: async () => {
   922→    if (typeof window === 'undefined') return;
   923→    // wait for hydration so the local library is loaded first
   924→    for (let i = 0; i < 100 && !useSimulator.getState().hydrated; i++) {
   925→      await new Promise((r) => setTimeout(r, 50));
   926→    }
   927→    let rows: { id: string; kind: string; data: unknown; updatedAt?: string }[] = [];
   928→    try {
   929→      const res = await fetch('/api/presets?full=1');
   930→      if (!res.ok) return;
   931→      rows = await res.json();
   932→    } catch {
   933→      return; // offline — the local library stands
   934→    }
   935→    if (!Array.isArray(rows)) return;
   936→    const lib = useSimulator.getState().presetLibrary;
   937→    let tables = [...lib.tables];
   938→    let fonts = [...lib.fonts];
   939→    let changed = false;
   940→    const seen = new Set<string>();
   941→    for (const row of rows) {
   942→      if (!row || typeof row.id !== 'string') continue;
   943→      if (row.kind !== 'table' && row.kind !== 'font') continue;
   944→      seen.add(row.id);
   945→      if (row.kind === 'table') {
   946→        const def = validateLibraryTablePreset(row.data);
   947→        if (!def) continue;
   948→        const existing = tables.find((t) => t.id === row.id);
   949→        if (!existing) {
   950→          tables.push(def);
   951→          changed = true;
   952→        } else if ((existing.updatedAt ?? '') < (def.updatedAt ?? row.updatedAt ?? '')
   953→          && JSON.stringify(existing) !== JSON.stringify(def)) {
   954→          tables = tables.map((t) => (t.id === row.id ? def : t));
   955→          changed = true;
   956→        }
   957→      } else {
   958→        const def = validateLibraryFontPreset(row.data);
   959→        if (!def) continue;
   960→        const existing = fonts.find((f) => f.id === row.id);
   961→        if (!existing) {
   962→          fonts.push(def);
   963→          changed = true;
   964→        } else if ((existing.updatedAt ?? '') < (def.updatedAt ?? row.updatedAt ?? '')
   965→          && JSON.stringify(existing) !== JSON.stringify(def)) {
   966→          fonts = fonts.map((f) => (f.id === row.id ? def : f));
   967→          changed = true;
   968→        }
   969→      }
   970→    }
   971→    // push local-only presets up to the server
   972→    for (const t of tables) if (!seen.has(t.id)) void apiUpsertPreset(t, 'table');
   973→    for (const f of fonts) if (!seen.has(f.id)) void apiUpsertPreset(f, 'font');
   974→    if (changed) {
   975→      const library = { tables, fonts };
   976→      useSimulator.setState({ presetLibrary: library });
   977→      persistLibrary(library);
   978→    }

  1135→/** Subscribe once from a client component to persist every change (debounced). */
  1136→export function attachPersistence() {
  1137→  if (typeof window === 'undefined') return () => {};
  1138→  let timer: ReturnType<typeof setTimeout> | null = null;
  1139→  const unsub = useSimulator.subscribe(() => {
  1140→    if (timer) clearTimeout(timer);
  1141→    timer = setTimeout(() => persist(useSimulator.getState()), 250);
  1142→  });
  1143→  return unsub;
  1144→}
