    41→class CaseChecker {
    42→  count = 0;
    43→
    44→  check(cond: boolean, msg: string): void {
    45→    this.count++;
    46→    if (!cond) throw new Error(`check failed: ${msg}`);
    47→  }
    48→
    49→  equal<T>(actual: T, expected: T, msg: string): void {
    50→    this.count++;
    51→    if (actual !== expected) {
    52→      throw new Error(`${msg}: expected ${JSON.stringify(expected)}, got ${JSON.stringify(actual)}`);
    53→    }
    54→  }
    55→}

    97→function validateDataReplica(fileData: string, title: string, c: CaseChecker): CvProjectJson {
    98→  const parsed = JSON.parse(fileData) as CvProjectJson;
    99→  c.check(
   100→    JSON.stringify(Object.keys(parsed)) === JSON.stringify(JSON_SCHEMA),
   101→    'top-level key set/order must exactly match the importer schema',
   102→  );
   103→  for (const scope of parsed.scopes) {
   104→    const keys = Object.keys(scope);
   105→    for (const key of SCOPE_SCHEMA) {
   106→      c.check(keys.includes(key), `scope must contain '${key}'`);
   107→    }
   108→  }
   109→  c.equal(parsed.name, title, 'project name must equal the requested title');
   110→  c.equal(parsed.timePeriod, 500, 'timePeriod');
   111→  c.equal(parsed.clockEnabled, true, 'clockEnabled');
   112→  c.check(/^[A-Za-z0-9]{20}$/.test(parsed.projectId), 'projectId must be 20-char alphanumeric');
   113→  const scope = parsed.scopes[0];
   114→  c.equal(parsed.scopes.length, 1, 'exactly one circuit');
   115→  c.check(typeof scope.id === 'number' && scope.id > 0, 'scope id must be a truthy number');
   116→  c.equal(parsed.focussedCircuit, scope.id, 'focussedCircuit === scope id');
   117→  c.equal(JSON.stringify(parsed.orderedTabs), JSON.stringify([String(scope.id)]), 'orderedTabs === [String(scope id)]');
   118→  c.equal(
   119→    JSON.stringify(scope.layout),
   120→    JSON.stringify({ width: 100, height: 100, title_x: 50, title_y: 13, titleEnabled: true }),
   121→    'scope layout',
   122→  );
   123→  c.equal(
   124→    JSON.stringify(scope.verilogMetadata),
   125→    JSON.stringify({ isVerilogCircuit: false, isMainCircuit: false, code: '// Write Some Verilog Code Here!', subCircuitScopeIds: [] }),
   126→    'scope verilogMetadata',
   127→  );
   128→  c.equal(JSON.stringify(scope.restrictedCircuitElementsUsed), '[]', 'restrictedCircuitElementsUsed empty');
   129→  c.equal(scope.name, 'Main', 'scope name');
   130→  c.check(
   131→    fileData === JSON.stringify(JSON.parse(fileData), null, 2),
   132→    'json must be pretty-printed with 2-space indent',
   133→  );
   134→  return parsed;
   135→}

   142→/** modules/AndGate.js input-pin creation loop, verbatim semantics. */
   143→function cvAndInputYs(n: number): number[] {
   144→  const ys: number[] = [];
   145→  if (n % 2 === 1) {
   146→    for (let i = 0; i < n / 2 - 1; i++) ys.push(-10 * (i + 1));
   147→    ys.push(0);
   148→    for (let i = n / 2 + 1; i < n; i++) ys.push(10 * (i + 1 - n / 2 - 1));
   149→  } else {
   150→    for (let i = 0; i < n / 2; i++) ys.push(-10 * (i + 1));
   151→    for (let i = n / 2; i < n; i++) ys.push(10 * (i + 1 - n / 2));
   152→  }
   153→  return ys;
   154→}
   155→
   156→/** modules/OrGate.js input-pin creation loop, verbatim semantics. */
   157→function cvOrInputYs(n: number): number[] {
   158→  const ys: number[] = [];
   159→  if (n % 2 === 1) {
   160→    for (let i = Math.floor(n / 2) - 1; i >= 0; i--) ys.push(-10 * (i + 1));
   161→    ys.push(0);
   162→    for (let i = 0; i < Math.floor(n / 2); i++) ys.push(10 * (i + 1));
   163→  } else {
   164→    for (let i = n / 2 - 1; i >= 0; i--) ys.push(-10 * (i + 1));
   165→    for (let i = 0; i < n / 2; i++) ys.push(10 * (i + 1));
   166→  }
   167→  return ys;
   168→}

   180→function componentsOf(scope: CvScopeJson): { type: string; comp: CvComponentJson }[] {
   181→  const out: { type: string; comp: CvComponentJson }[] = [];
   182→  for (const [key, value] of Object.entries(scope)) {
   183→    if (NON_MODULE_KEYS.has(key)) continue;
   184→    if (Array.isArray(value)) {
   185→      for (const comp of value as CvComponentJson[]) out.push({ type: key, comp });
   186→    }
   187→  }
   188→  return out;
   189→}

   191→function structuralChecks(project: CvProjectJson, inputBits: number, c: CaseChecker): CvScopeJson {
   192→  const scope = project.scopes[0];
   193→  const nodes = scope.allNodes;
   194→  const comps = componentsOf(scope);
   195→
   196→  // Module keys are all known types.
   197→  for (const key of Object.keys(scope)) {
   198→    if (!NON_MODULE_KEYS.has(key)) c.check(KNOWN_MODULES.has(key), `unknown module array '${key}'`);
   199→  }
   200→
   201→  // Connections: in range, mutual, and every type-2 node has exactly 2.
   202→  for (let i = 0; i < nodes.length; i++) {
   203→    const nd = nodes[i];
   204→    c.equal(nd.bitWidth, 1, `node ${i} bitWidth`);
   205→    c.check(Number.isInteger(nd.x) && nd.x % 10 === 0, `node ${i} x must be a multiple of 10`);
   206→    c.check(Number.isInteger(nd.y) && nd.y % 10 === 0, `node ${i} y must be a multiple of 10`);
   207→    c.check([0, 1, 2].includes(nd.type), `node ${i} type`);
   208→    for (const j of nd.connections) {
   209→      c.check(Number.isInteger(j) && j >= 0 && j < nodes.length, `node ${i} connection ${j} in range`);
   210→      c.check(nodes[j].connections.includes(i), `connection ${i}->${j} must be mutual`);
   211→    }
   212→    if (nd.type === 2) c.equal(nd.connections.length, 2, `type-2 node ${i} must have exactly 2 connections`);
   213→  }
   214→
   215→  // scope.nodes === sorted indices of all type-2 nodes.
   216→  const type2 = nodes.map((nd, i) => (nd.type === 2 ? i : -1)).filter((i) => i >= 0);
   217→  c.equal(JSON.stringify(scope.nodes), JSON.stringify(type2), 'scope.nodes lists exactly the type-2 indices (sorted)');
   218→
   219→  // Every type-0/1 node claimed by exactly one component pin slot.
   220→  const claimed = new Array<number>(nodes.length).fill(0);
   221→  const claim = (idx: unknown, where: string): void => {
   222→    c.check(typeof idx === 'number' && Number.isInteger(idx) && idx >= 0 && idx < nodes.length, `${where} references a valid node index`);
   223→    if (typeof idx === 'number') claimed[idx]++;
   224→  };
   225→
   226→  const inputLabels: string[] = [];
   227→  const inputIds: string[] = [];
   228→  let sevenSegCount = 0;
   229→
   230→  for (const { type, comp } of comps) {
   231→    c.check(comp.objectType === type, `component objectType '${comp.objectType}' matches its array '${type}'`);
   232→    c.equal(typeof comp.label, 'string', `${type} label is a string`);
   233→    c.equal(comp.direction, 'RIGHT', `${type} direction`);
   234→    c.equal(comp.labelDirection, 'LEFT', `${type} labelDirection`);
   235→    c.check(typeof comp.propagationDelay === 'number', `${type} propagationDelay is a number`);
   236→    c.check(comp.x % 10 === 0 && comp.y % 10 === 0, `${type} at (${comp.x},${comp.y}) on the 10-grid`);
   237→    c.check(Array.isArray(comp.customData.constructorParamaters), `${type} constructorParamaters array`);
   238→    const cp = comp.customData.constructorParamaters;
   239→    const nodeKeys = Object.keys(comp.customData.nodes).sort();
   240→
   241→    if (type === 'Input') {
   242→      c.equal(JSON.stringify(nodeKeys), JSON.stringify(['output1']), 'Input node keys');
   243→      c.equal(cp[0], 'RIGHT', 'Input direction param');
   244→      c.equal(cp[1], 1, 'Input bitWidth param');
   245→      const layout = cp[2] as { x: number; y: number; id: string };
   246→      c.check(!!layout && typeof layout.id === 'string' && /^[A-Za-z0-9]{20}$/.test(layout.id), 'Input layoutProperties id 20-char alnum');
   247→      inputIds.push(layout.id);
   248→      c.equal(layout.x, 0, 'Input layoutProperties x');
   249→      const out = comp.customData.nodes.output1 as number;
   250→      claim(out, 'Input output1');
   251→      c.equal(nodes[out].x, 10, 'Input output1 pin offset x (bitWidth*10)');
   252→      c.equal(nodes[out].y, 0, 'Input output1 pin offset y');
   253→      c.equal(nodes[out].type, 1, 'Input output1 node type');
   254→      c.equal(comp.propagationDelay, 0, 'Input propagationDelay 0');
   255→      c.equal(comp.customData.values?.state, 0, 'Input values.state');
   256→      inputLabels.push(comp.label as string);
   257→    } else if (type === 'ConstantVal') {
   258→      c.equal(JSON.stringify(nodeKeys), JSON.stringify(['output1']), 'ConstantVal node keys');
   259→      c.equal(cp[0], 'RIGHT', 'ConstantVal direction param');
   260→      c.equal(cp[1], 1, 'ConstantVal bitWidth param');
   261→      c.check(cp[2] === '0' || cp[2] === '1', 'ConstantVal state must be the STRING "0" or "1"');
   262→      const out = comp.customData.nodes.output1 as number;
   263→      claim(out, 'ConstantVal output1');
   264→      c.equal(nodes[out].x, 10, 'ConstantVal output1 pin offset x');
   265→      c.equal(nodes[out].y, 0, 'ConstantVal output1 pin offset y');
   266→      c.equal(nodes[out].type, 1, 'ConstantVal output1 node type');
   267→      c.equal(comp.propagationDelay, 0, 'ConstantVal propagationDelay 0');
   268→    } else if (type === 'NotGate') {
   269→      c.equal(JSON.stringify(nodeKeys), JSON.stringify(['inp1', 'output1']), 'NotGate node keys');
   270→      c.equal(JSON.stringify(cp), JSON.stringify(['RIGHT', 1]), 'NotGate constructorParamaters');
   271→      const inp = comp.customData.nodes.inp1 as number;
   272→      const out = comp.customData.nodes.output1 as number;
   273→      claim(inp, 'NotGate inp1');
   274→      claim(out, 'NotGate output1');
   275→      c.equal(nodes[inp].x, -10, 'NotGate inp1 offset x');
   276→      c.equal(nodes[inp].y, 0, 'NotGate inp1 offset y');
   277→      c.equal(nodes[inp].type, 0, 'NotGate inp1 node type');
   278→      c.equal(nodes[out].x, 20, 'NotGate output1 offset x');
   279→      c.equal(nodes[out].y, 0, 'NotGate output1 offset y');
   280→      c.equal(nodes[out].type, 1, 'NotGate output1 node type');
   281→      c.equal(comp.propagationDelay, 10, 'NotGate propagationDelay 10');
   282→    } else if (type === 'AndGate' || type === 'OrGate') {
   283→      c.equal(JSON.stringify(nodeKeys), JSON.stringify(['inp', 'output1']), `${type} node keys`);
   284→      const inputSize = cp[1] as number;
   285→      c.equal(cp[0], 'RIGHT', `${type} direction param`);
   286→      c.equal(cp[2], 1, `${type} bitWidth param`);
   287→      const inp = comp.customData.nodes.inp as number[];
   288→      c.check(Array.isArray(inp), `${type} inp is an array`);
   289→      c.equal(inp.length, inputSize, `${type} inp length matches inputSize`);
   290→      const ys = type === 'AndGate' ? cvAndInputYs(inputSize) : cvOrInputYs(inputSize);
   291→      inp.forEach((idx, k) => {
   292→        claim(idx, `${type} inp[${k}]`);
   293→        c.equal(nodes[idx].x, -10, `${type} inp[${k}] offset x`);
   294→        c.equal(nodes[idx].y, ys[k], `${type} inp[${k}] offset y (constructor order)`);
   295→        c.equal(nodes[idx].type, 0, `${type} inp[${k}] node type`);
   296→      });
   297→      const out = comp.customData.nodes.output1 as number;
   298→      claim(out, `${type} output1`);
   299→      c.equal(nodes[out].x, 20, `${type} output1 offset x`);
   300→      c.equal(nodes[out].y, 0, `${type} output1 offset y`);
   301→      c.equal(nodes[out].type, 1, `${type} output1 node type`);
   302→      c.equal(comp.propagationDelay, 10, `${type} propagationDelay 10`);
   303→      if (type === 'AndGate') c.check(inputSize >= 2 && inputSize <= 5, `AndGate inputSize ${inputSize} within 2..5`);
   304→      else c.check(inputSize >= 2 && inputSize <= 8, `OrGate inputSize ${inputSize} within 2..8`);
   305→    } else if (type === 'SevenSegDisplay') {
   306→      sevenSegCount++;
   307→      c.equal(JSON.stringify(nodeKeys), JSON.stringify(['a', 'b', 'c', 'd', 'dot', 'e', 'f', 'g']), 'SevenSegDisplay node keys (all 8 pins)');
   308→      c.equal(JSON.stringify(cp), JSON.stringify(['Red']), 'SevenSegDisplay constructorParamaters');
   309→      c.equal(comp.propagationDelay, 10, 'SevenSegDisplay propagationDelay 10');
   310→      c.check(!!comp.subcircuitMetadata, 'SevenSegDisplay has subcircuitMetadata');
   311→      for (const [key, [ex, ey]] of Object.entries(CV_SEVENSEG_OFFSETS)) {
   312→        const idx = comp.customData.nodes[key] as number;
   313→        claim(idx, `SevenSegDisplay ${key}`);
   314→        c.equal(nodes[idx].x, ex, `SevenSegDisplay ${key} offset x`);
   315→        c.equal(nodes[idx].y, ey, `SevenSegDisplay ${key} offset y`);
   316→        c.equal(nodes[idx].type, 0, `SevenSegDisplay ${key} node type`);
   317→      }
   318→    }
   319→  }
   320→
   321→  for (let i = 0; i < nodes.length; i++) {
   322→    if (nodes[i].type === 2) {
   323→      c.equal(claimed[i], 0, `type-2 node ${i} must not be claimed by a component`);
   324→    } else {
   325→      c.equal(claimed[i], 1, `pin node ${i} (type ${nodes[i].type}) must be claimed by exactly one component slot`);
   326→    }
   327→  }
   328→
   329→  c.equal(sevenSegCount, 1, 'exactly one SevenSegDisplay');
   330→  c.equal(inputLabels.length, inputBits, 'Input count equals input width');
   331→  c.equal(new Set(inputLabels).size, inputLabels.length, 'Input labels unique');
   332→  c.equal(
   333→    JSON.stringify([...inputLabels].sort()),
   334→    JSON.stringify([...inputVarNames(inputBits as InputWidth)].sort()),
   335→    'Input labels are the variable names',
   336→  );
   337→  c.equal(new Set(inputIds).size, inputIds.length, 'Input layoutProperties ids unique');
   338→  return scope;
   339→}

   345→interface GateJson { inNodes: number[]; outNode: number; kind: 'not' | 'and' | 'or'; }
   346→
   347→class MiniSim {
   348→  private readonly parent: number[] = [];
   349→  private readonly inputs: { outNode: number; label: string }[] = [];
   350→  private readonly consts: { outNode: number; state: string }[] = [];
   351→  private readonly gates: GateJson[] = [];
   352→  private readonly display: Record<string, number> | null = null;
   353→  readonly driverErrors: string[] = [];
   354→
   355→  constructor(scope: CvScopeJson, private readonly inputBits: number) {
   356→    const nodes = scope.allNodes;
   357→    this.parent = nodes.map((_, i) => i);
   358→    for (let i = 0; i < nodes.length; i++) {
   359→      for (const j of nodes[i].connections) this.union(i, j);
   360→    }
   361→
   362→    // One driver (output pin) per net — output-output shorts are import bugs.
   363→    const driversPerNet = new Map<number, number>();
   364→    const noteDriver = (nodeIdx: number, what: string): void => {
   365→      const root = this.find(nodeIdx);
   366→      driversPerNet.set(root, (driversPerNet.get(root) ?? 0) + 1);
   367→      if ((driversPerNet.get(root) ?? 0) > 1) this.driverErrors.push(`${what} shorts another driver on net ${root}`);
   368→    };
   369→
   370→    for (const { type, comp } of componentsOf(scope)) {
   371→      const cd = comp.customData;
   372→      if (type === 'Input') {
   373→        this.inputs.push({ outNode: cd.nodes.output1 as number, label: comp.label as string });
   374→        noteDriver(cd.nodes.output1 as number, 'Input');
   375→      } else if (type === 'ConstantVal') {
   376→        this.consts.push({ outNode: cd.nodes.output1 as number, state: cd.constructorParamaters[2] as string });
   377→        noteDriver(cd.nodes.output1 as number, 'ConstantVal');
   378→      } else if (type === 'NotGate') {
   379→        this.gates.push({ inNodes: [cd.nodes.inp1 as number], outNode: cd.nodes.output1 as number, kind: 'not' });
   380→        noteDriver(cd.nodes.output1 as number, 'NotGate');
   381→      } else if (type === 'AndGate') {
   382→        this.gates.push({ inNodes: cd.nodes.inp as number[], outNode: cd.nodes.output1 as number, kind: 'and' });
   383→        noteDriver(cd.nodes.output1 as number, 'AndGate');
   384→      } else if (type === 'OrGate') {
   385→        this.gates.push({ inNodes: cd.nodes.inp as number[], outNode: cd.nodes.output1 as number, kind: 'or' });
   386→        noteDriver(cd.nodes.output1 as number, 'OrGate');
   387→      } else if (type === 'SevenSegDisplay') {
   388→        this.display = cd.nodes as unknown as Record<string, number>;
   389→      }
   390→    }
   391→  }
   392→
   393→  private find(i: number): number {
   394→    while (this.parent[i] !== i) {
   395→      this.parent[i] = this.parent[this.parent[i]];
   396→      i = this.parent[i];
   397→    }
   398→    return i;
   399→  }
   400→
   401→  private union(a: number, b: number): void {
   402→    const ra = this.find(a);
   403→    const rb = this.find(b);
   404→    if (ra !== rb) this.parent[rb] = ra;
   405→  }
   406→
   407→  /** Simulate one input code and read the eight display pins. */
   408→  run(code: number): Record<SegmentName, 0 | 1> {
   409→    const value = new Map<number, 0 | 1>();
   410→    const varNames = inputVarNames(this.inputBits as InputWidth);
   411→    for (const inp of this.inputs) {
   412→      const bitPos = this.inputBits - 1 - varNames.indexOf(inp.label);
   413→      value.set(this.find(inp.outNode), ((code >> bitPos) & 1) as 0 | 1);
   414→    }
   415→    for (const cst of this.consts) {
   416→      value.set(this.find(cst.outNode), parseInt(cst.state, 2) as 0 | 1);
   417→    }
   418→
   419→    // Combinational propagation to fixpoint (bounded by gate count).
   420→    let progress = true;
   421→    let guard = 0;
   422→    while (progress && guard++ < this.gates.length + 4) {
   423→      progress = false;
   424→      for (const gate of this.gates) {
   425→        const outNet = this.find(gate.outNode);
   426→        if (value.has(outNet)) continue;
   427→        let acc: 0 | 1 | null = null;
   428→        let allDefined = true;
   429→        for (const inNode of gate.inNodes) {
   430→          const v = value.get(this.find(inNode));
   431→          if (v === undefined) { allDefined = false; break; }
   432→          if (gate.kind === 'not') acc = v === 1 ? 0 : 1;
   433→          else if (acc === null) acc = v;
   434→          else if (gate.kind === 'and') acc = acc === 1 && v === 1 ? 1 : 0;
   435→          else acc = acc === 1 || v === 1 ? 1 : 0;
   436→        }
   437→        if (allDefined && acc !== null) {
   438→          value.set(outNet, acc);
   439→          progress = true;
   440→        }
   441→      }
   442→    }
   443→
   444→    const pinToSegment: Record<string, SegmentName> = {
   445→      a: 'A', b: 'B', c: 'C', d: 'D', e: 'E', f: 'F', g: 'G', dot: 'DP',
   446→    };
   447→    const out = {} as Record<SegmentName, 0 | 1>;
   448→    for (const [pin, seg] of Object.entries(pinToSegment)) {
   449→      const nodeIdx = this.display?.[pin];
   450→      out[seg] = nodeIdx === undefined ? 0 : (value.get(this.find(nodeIdx)) ?? 0);
   451→    }
   452→    return out;
   453→  }
   454→}

   456→/* ------------------------------------------------------------------ */
   457→/* SOP reference evaluation                                            */
   458→/* ------------------------------------------------------------------ */
   459→
   460→function referenceTerms(engines: SegmentEngineMap, seg: SegmentName, logicSource: 'minimized' | 'direct', inputBits: number): Term[] {
   461→  return logicSource === 'direct'
   462→    ? engines[seg].ones.map((m) => implicantToSopTerm({ bits: m, mask: (1 << inputBits) - 1 }, inputBits))
   463→    : engines[seg].activeSopTerms;
   464→}
   465→
   466→function evalSop(terms: Term[], code: number, inputBits: number): 0 | 1 {
   467→  const varNames = inputVarNames(inputBits as InputWidth);
   468→  for (const term of terms) {
   469→    if (term.literals.length === 0) return 1;
   470→    let match = true;
   471→    for (const lit of term.literals) {
   472→      const bit = (code >> (inputBits - 1 - varNames.indexOf(lit.name))) & 1;
   473→      if (bit !== (lit.complemented ? 0 : 1)) { match = false; break; }
   474→    }
   475→    if (match) return 1;
   476→  }
   477→  return 0;
   478→}

   480→/* ------------------------------------------------------------------ */
   481→/* Case runner                                                         */
   482→/* ------------------------------------------------------------------ */
   483→
   484→interface CaseCtx {
   485→  engines: SegmentEngineMap;
   486→  result: CircuitverseBuildResult;
   487→  stats: CircuitverseStats;
   488→  scope: CvScopeJson;
   489→}
   490→
   491→function runCase(
   492→  name: string,
   493→  inputBits: InputWidth,
   494→  table: TruthTable,
   495→  logicSource: 'minimized' | 'direct',
   496→  extra?: (ctx: CaseCtx, c: CaseChecker) => void,
   497→): { pass: boolean; checks: number } {
   498→  const c = new CaseChecker();
   499→  const title = `Verify ${name}`;
   500→  try {
   501→    const engines = computeEngines(table, {}, {});
   502→    const result = buildCircuitverseProject(inputBits, engines, { title, logicSource });
   503→    const project = validateDataReplica(result.json, title, c);
   504→    const scope = structuralChecks(project, inputBits, c);
   505→
   506→    // Stats agree with the serialized file.
   507→    const comps = componentsOf(scope);
   508→    const count = (t: string): number => comps.filter((x) => x.type === t).length;
   509→    c.equal(result.stats.inputs, count('Input'), 'stats.inputs vs file');
   510→    c.equal(result.stats.inverters, count('NotGate'), 'stats.inverters vs file');
   511→    c.equal(result.stats.andGates, count('AndGate'), 'stats.andGates vs file');
   512→    c.equal(result.stats.orGates, count('OrGate'), 'stats.orGates vs file');
   513→    c.equal(result.stats.constants, count('ConstantVal'), 'stats.constants vs file');
   514→    c.equal(result.stats.intermediates, scope.allNodes.filter((nd) => nd.type === 2).length, 'stats.intermediates vs file');
   515→    const edgeCount = scope.allNodes.reduce((s, nd) => s + nd.connections.length, 0) / 2;
   516→    c.equal(result.stats.wires, edgeCount, 'stats.wires vs file');
   517→    c.equal(result.stats.segments.length, 8, 'stats has 8 segment entries');
   518→    c.equal(
   519→      JSON.stringify(result.stats.segments.map((s) => s.segment)),
   520→      JSON.stringify([...SEGMENT_NAMES]),
   521→      'segment stats in SEGMENT_NAMES order',
   522→    );
   523→
   524→    // Simulate every input code against the SOP reference and the truth table.
   525→    const sim = new MiniSim(scope, inputBits);
   526→    c.equal(sim.driverErrors.length, 0, `every net has at most one driver (${sim.driverErrors.join('; ')})`);
   527→    const refTerms = {} as Record<SegmentName, Term[]>;
   528→    for (const seg of SEGMENT_NAMES) refTerms[seg] = referenceTerms(engines, seg, logicSource, inputBits);
   529→    for (let code = 0; code < (1 << inputBits); code++) {
   530→      const segVals = sim.run(code);
   531→      for (const seg of SEGMENT_NAMES) {
   532→        c.equal(segVals[seg], evalSop(refTerms[seg], code, inputBits), `code ${code} segment ${seg}: circuit vs SOP reference`);
   533→        const cell = table.rows[code].outputs[seg];
   534→        if (cell === 0 || cell === 1) {
   535→          c.equal(segVals[seg], cell, `code ${code} segment ${seg}: circuit vs truth table`);
   536→        }
   537→      }
   538→    }
   539→
   540→    extra?.({ engines, result, stats: result.stats, scope }, c);
   541→    console.log(`PASS ${name} — ${c.count} checks`);
   542→    return { pass: true, checks: c.count };
   543→  } catch (err) {
   544→    const msg = err instanceof Error ? err.message : String(err);
   545→    console.error(`FAIL ${name} — aborted after ${c.count} checks: ${msg}`);
   546→    return { pass: false, checks: c.count };
   547→  }
   548→}
