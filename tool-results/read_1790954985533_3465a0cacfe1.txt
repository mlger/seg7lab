   360→    /**
   361→     * function to resolve a node
   362→     */
   363→    resolve() {
   364→        if (this.type == NODE_OUTPUT) {
   365→            // Since output node forces its value on its neighbours, remove its contentions.
   366→            // An existing contention will now trickle to the other output node that was causing
   367→            // the contention.
   368→            simulationArea.contentionPending.removeAllContentionsForNode(this);
   369→        }
   370→        // Remove Propogation of values (TriState)
   371→        if (this.value == undefined) {
   372→            for (var i = 0; i < this.connections.length; i++) {
   373→                if (this.connections[i].value !== undefined) {
   374→                    this.connections[i].value = undefined;
   375→                    simulationArea.simulationQueue.add(this.connections[i]);
   376→                }
   377→            }
   378→
   379→            if (this.type == NODE_INPUT) {
   380→                if (this.parent.objectType == 'Splitter') {
   381→                    this.parent.removePropagation();
   382→                } else
   383→                    if (this.parent.isResolvable()) { simulationArea.simulationQueue.add(this.parent); }
   384→                    else {
   385→                        this.parent.setOutputsUpstream(false);
   386→                        this.parent.removePropagation();
   387→                    }
   388→            }
   389→
   390→            if (this.type == NODE_OUTPUT && !this.subcircuitOverride) {
   391→                if (this.parent.isResolvable() && !this.parent.queueProperties.inQueue) {
   392→                    if (this.parent.objectType == 'TriState' || this.parent.objectType == 'ControlledInverter') {
   393→                        if (this.parent.state.value) { simulationArea.simulationQueue.add(this.parent); }
   394→                    } else {
   395→                        simulationArea.simulationQueue.add(this.parent);
   396→                    }
   397→                }
   398→            }
   399→
   400→            return;
   401→        }
   402→
   403→        // For input nodes, resolve its parents if they are resolvable at this point.
   404→        if (this.type == NODE_INPUT) {
   405→            if (this.parent.isResolvable()) { simulationArea.simulationQueue.add(this.parent); }
   406→            else { this.parent.setOutputsUpstream(false); }
   407→        }
   408→
   409→        for (var i = 0; i < this.connections.length; i++) {
   410→            const node = this.connections[i];
   411→
   412→            switch (node.type) {
   413→            case NODE_OUTPUT:
   414→                // If node value is differnet and node value is upstream, then contention.
   415→                if ((node.isValueUpstream && node.value != this.value) || node.bitWidth != this.bitWidth) {
   416→                    // Check contentions
   417→                    if (node.value != undefined && node.parent.objectType != 'SubCircuit'
   418→                        && !(node.subcircuitOverride && node.scope != this.scope)) {
   419→                        
   420→                        simulationArea.contentionPending.add(node, this);
   421→                        break;
   422→                    }
   423→                } else {
   424→                    // Output node was given an agreeing value, so remove any contention
   425→                    // entry between these two nodes if it exists.
   426→                    simulationArea.contentionPending.remove(node, this);
   427→                }
   428→
   429→            // Fallthrough. NODE_OUTPUT propagates like a contention checked NODE_INPUT
   430→            case NODE_INPUT:
   431→                // Check bitwidths
   432→                if (this.bitWidth != node.bitWidth) {
   433→                    this.highlighted = true;
   434→                    node.highlighted = true;
   435→                    showError(`BitWidth Error: ${this.bitWidth} and ${node.bitWidth}`);
   436→                    break;
   437→                }
   438→
   439→            // Fallthrough. NODE_INPUT propagates like a bitwidth checked NODE_INTERMEDIATE
   440→            case NODE_INTERMEDIATE:
   441→
   442→                if (node.value != this.value || node.bitWidth != this.bitWidth) {
   443→                    // Propagate
   444→                    node.bitWidth = this.bitWidth;
   445→                    node.value = this.value;
   446→                    simulationArea.simulationQueue.add(node);
   447→                }
   448→            default:
   449→                break;
   450→            }
   451→        }
   452→    }

   613→                for (var i = 0; i < simulationArea.multipleObjectSelections.length; i++) {
   614→                    simulationArea.multipleObjectSelections[i].drag();
   615→                }
   616→            }
   617→            if (this.type == NODE_INTERMEDIATE) {
   618→                if (this.connections.length == 1 && this.connections[0].absX() == simulationArea.mouseX && this.absX() == simulationArea.mouseX) {
   619→                    this.y = simulationArea.mouseY - this.parent.y;
   620→                    this.prev = 'a';
   621→                    return;
   622→                } if (this.connections.length == 1 && this.connections[0].absY() == simulationArea.mouseY && this.absY() == simulationArea.mouseY) {
   623→                    this.x = simulationArea.mouseX - this.parent.x;
   624→                    this.prev = 'a';
   625→                    return;
   626→                }
   627→                if (this.connections.length == 1 && this.connections[0].absX() == this.absX() && this.connections[0].absY() == this.absY()) {
   628→                    this.connections[0].clicked = true;
   629→                    this.connections[0].wasClicked = true;
   630→                    simulationArea.lastSelected = this.connections[0];
   631→                    this.delete();
   632→                    return;
   633→                }
   634→            }
   635→
   636→            if (this.prev == 'a' && distance(simulationArea.mouseX, simulationArea.mouseY, this.absX(), this.absY()) >= 10) {
   637→                if (Math.abs(this.x + this.parent.x - simulationArea.mouseX) > Math.abs(this.y + this.parent.y - simulationArea.mouseY)) {
   638→                    this.prev = 'x';
   639→                } else {
   640→                    this.prev = 'y';
   641→                }
   642→            } else if (this.prev == 'x' && this.absY() == simulationArea.mouseY) {
   643→                this.prev = 'a';
   644→            } else if (this.prev == 'y' && this.absX() == simulationArea.mouseX) {
   645→                this.prev = 'a';
   646→            }
   647→        } else if (this.wasClicked && !this.clicked) {
   648→            this.wasClicked = false;
   649→
   650→            if (simulationArea.mouseX == this.absX() && simulationArea.mouseY == this.absY()) {
   651→                return; // no new node situation
   652→            }
   653→
   654→            var x1; var y1; var x2; var y2; var
   655→                flag = 0;
   656→            var n1; var
   657→                n2;
   658→
   659→            // (x,y) present node, (x1,y1) node 1 , (x2,y2) node 2
   660→            // n1 - node 1, n2 - node 2
   661→            // node 1 may or may not be there
   662→            // flag = 0  - node 2 only
   663→            // flag = 1  - node 1 and node 2
   664→            x2 = simulationArea.mouseX;
   665→            y2 = simulationArea.mouseY;
   666→            const x = this.absX();
   667→            const y = this.absY();
   668→
   669→            if (x != x2 && y != y2) {
   670→                // Rare Exception Cases
   671→                if (this.prev == 'a' && distance(simulationArea.mouseX, simulationArea.mouseY, this.absX(), this.absY()) >= 10) {
   672→                    if (Math.abs(this.x + this.parent.x - simulationArea.mouseX) > Math.abs(this.y + this.parent.y - simulationArea.mouseY)) {
   673→                        this.prev = 'x';
   674→                    } else {
   675→                        this.prev = 'y';
   676→                    }
   677→                }
   678→
   679→                flag = 1;
   680→                if (this.prev == 'x') {
   681→                    x1 = x2;
   682→                    y1 = y;
   683→                } else if (this.prev == 'y') {
   684→                    y1 = y2;
   685→                    x1 = x;
   686→                }
   687→            }
   688→
   689→            if (flag == 1) {
   690→                for (var i = 0; i < this.parent.scope.allNodes.length; i++) {
   691→                    if (x1 == this.parent.scope.allNodes[i].absX() && y1 == this.parent.scope.allNodes[i].absY()) {
   692→                        n1 = this.parent.scope.allNodes[i];
   693→                        break;
   694→                    }
   695→                }
   696→
   697→                if (n1 == undefined) {
   698→                    n1 = new Node(x1, y1, 2, this.scope.root);
   699→                    for (var i = 0; i < this.parent.scope.wires.length; i++) {
   700→                        if (this.parent.scope.wires[i].checkConvergence(n1)) {
   701→                            this.parent.scope.wires[i].converge(n1);
   702→                            break;
   703→                        }
   704→                    }
   705→                }
   706→                this.connect(n1);
   707→            }
   708→
   709→            for (var i = 0; i < this.parent.scope.allNodes.length; i++) {
   710→                if (x2 == this.parent.scope.allNodes[i].absX() && y2 == this.parent.scope.allNodes[i].absY()) {
   711→                    n2 = this.parent.scope.allNodes[i];
   712→                    break;
   713→                }
   714→            }
   715→
   716→            if (n2 == undefined) {
   717→                n2 = new Node(x2, y2, 2, this.scope.root);
   718→                for (var i = 0; i < this.parent.scope.wires.length; i++) {
   719→                    if (this.parent.scope.wires[i].checkConvergence(n2)) {
   720→                        this.parent.scope.wires[i].converge(n2);
   721→                        break;
   722→                    }
   723→                }
   724→            }
   725→            if (flag == 0) this.connect(n2);
   726→            else n1.connect(n2);
   727→            if (simulationArea.lastSelected == this) simulationArea.lastSelected = n2;
   728→        }
   729→
   730→        if (this.type == NODE_INTERMEDIATE && simulationArea.mouseDown == false) {
   731→            if (this.connections.length == 2) {
   732→                if ((this.connections[0].absX() == this.connections[1].absX()) || (this.connections[0].absY() == this.connections[1].absY())) {
   733→                    this.connections[0].connect(this.connections[1]);
   734→                    this.delete();
   735→                }
   736→            } else if (this.connections.length == 0) this.delete();
   737→        }
   738→    }
   739→
   740→    /**
   741→     * function delete a node
   742→     */
   743→    delete() {
   744→        updateSimulationSet(true);
   745→        this.deleted = true;
   746→        this.parent.scope.allNodes.clean(this);
   747→        this.parent.scope.nodes.clean(this);
   748→
   749→        this.parent.scope.root.nodeList.clean(this); // Hope this works! - Can cause bugs
   750→
   751→        if (simulationArea.lastSelected == this) simulationArea.lastSelected = undefined;
   752→        for (var i = 0; i < this.connections.length; i++) {
   753→            this.connections[i].connections.clean(this);
   754→            this.connections[i].checkDeleted();
   755→        }
   756→
   757→        this.scope.timeStamp = new Date().getTime();
   758→
   759→        wireToBeCheckedSet(1);
   760→        forceResetNodesSet(true);
   761→        scheduleUpdate();
   762→    }
   763→
   764→    isClicked() {
   765→        return this.absX() == simulationArea.mouseX && this.absY() == simulationArea.mouseY;
   766→    }
   767→
   768→    isHover() {
   769→        return this.absX() == simulationArea.mouseX && this.absY() == simulationArea.mouseY;
   770→    }

   772→    /**
   773→     * if input nodde: it resolves the parent
   774→     * else: it adds all the nodes onto the stack
   775→     * and they are processed to generate verilog
   776→     */
   777→    nodeConnect() {
   778→        var x = this.absX();
   779→        var y = this.absY();
   780→        var n;
   781→
   782→        for (var i = 0; i < this.parent.scope.allNodes.length; i++) {
   783→            if (this != this.parent.scope.allNodes[i] && x == this.parent.scope.allNodes[i].absX() && y == this.parent.scope.allNodes[i].absY()) {
   784→                n = this.parent.scope.allNodes[i];
   785→                if (this.type == NODE_INTERMEDIATE) {
   786→                    for (var j = 0; j < this.connections.length; j++) {
   787→                        n.connect(this.connections[j]);
   788→                    }
   789→                    this.delete();
   790→                } else {
   791→                    this.connect(n);
   792→                }
   793→
   794→                break;
   795→            }
   796→        }
   797→
   798→        if (n == undefined) {
   799→            for (var i = 0; i < this.parent.scope.wires.length; i++) {
   800→                if (this.parent.scope.wires[i].checkConvergence(this)) {
   801→                    var n = this;
   802→                    if (this.type != NODE_INTERMEDIATE) {
   803→                        n = new Node(this.absX(), this.absY(), 2, this.scope.root);
   804→                        this.connect(n);
   805→                    }
   806→                    this.parent.scope.wires[i].converge(n);
   807→                    break;
   808→                }
   809→            }
   810→        }
   811→    }
   812→
   813→    processVerilog() {
   814→        if (this.type == NODE_INPUT) {
   815→            if (this.parent.isVerilogResolvable()) { this.scope.stack.push(this.parent); }
   816→        }
   817→
   818→        for (var i = 0; i < this.connections.length; i++) {
   819→            if (this.connections[i].verilogLabel != this.verilogLabel) {
   820→                this.connections[i].verilogLabel = this.verilogLabel;
   821→                this.scope.stack.push(this.connections[i]);
   822→            }
   823→        }
   824→    }
