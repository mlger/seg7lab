     1→/**
     2→ * Truth table: auto-generated for the selected input width, fully editable
     3→ * (click an output cell to cycle 0 → 1 → X, or click a segment directly on the
     4→ * row's preview glyph). Clicking a row selects that input.
     5→ *
     6→ * Features:
     7→ *  - Live mini 7-segment preview of every row (derived from the table itself),
     8→ *    shown after the Hex column; its segments are clickable to toggle cells.
     9→ *  - Drag a row's preview glyph onto another row to copy its output pattern —
    10→ *    inputs are excluded, they always stay untouched.
    11→ *  - Input variables default to A B C D / A B C D E (A = MSB) and can be
    12→ *    renamed by clicking a variable header — names flow through expressions,
    13→ *    K-map, circuit and every export.
    14→ *  - Row order can start from any code (e.g. ABCD = 0001, wrapping around).
    15→ *  - Shift tools: rotate one row's values, one segment's column, or the whole
    16→ *    table (all rows / all columns) — with one-level undo.
    17→ *  - FONT preset window controls (which glyph starts the mapping).
    18→ */

    41→function cellColor(v: SegmentOutput): string {
    42→  return v === 1
    43→    ? 'border-emerald-500/40 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400'
    44→    : v === 'X'

    50→ * One memoized truth-table row. Extracted from the map so drag feedback
    51→ * (dropRow changes) and other parent re-renders never re-render the other
    52→ * 15–31 rows: every prop is a primitive, a stable ref from the immutable
    53→ * store, or a useCallback handler, so React.memo's shallow compare holds.

    55→interface TruthRowProps {
    56→  row: TruthTableRow;
    57→  inputBits: InputWidth;
    58→  /** Character label for the Char column (font glyph / hex digit / null). */

    60→  active: boolean;

    71→  onRowHover: (input: number) => void;
    72→  onRowLeave: (input: number) => void;

    75→const TruthRow = memo(function TruthRow({
    76→  row, inputBits, char, active, isDragSource, isDropTarget,
    77→  registerRef, onSelectRow, onCycleCell, onTogglePreviewSegment,
    78→  onDragRowStart, onDragRowEnd, onRowDrop, onRowHover, onRowLeave,
    79→}: TruthRowProps) {
    80→  const bits = row.input.toString(2).padStart(inputBits, '0');
    81→  return (
    82→    <TableRow
    83→      ref={(el) => {
    84→        registerRef(row.input, el);
    85→      }}
    86→      className={cn(
    87→        'cursor-pointer',
    88→        active ? 'bg-primary/10' : 'hover:bg-muted/50',
    89→        isDragSource && 'opacity-50',
    90→        isDropTarget && 'bg-primary/15 ring-2 ring-inset ring-primary/50',
    91→      )}
    92→      onClick={() => onSelectRow(row.input)}
    93→      aria-label={`Select input row ${row.input}`}
    94→      onDragOver={(e) => {
    95→        if (!e.dataTransfer.types.includes('text/x-sevenseg-row')) return;
    96→        e.preventDefault();
    97→        e.dataTransfer.dropEffect = 'copy';
    98→        onRowHover(row.input);
    99→      }}
   100→      onDragLeave={(e) => {
   101→        if (!e.currentTarget.contains(e.relatedTarget as Node | null)) {
   102→          onRowLeave(row.input);
   103→        }
   104→      }}
   105→      onDrop={(e) => onRowDrop(e, row.input)}
   106→    >
   107→      {[...bits].map((b, i) => (
   108→        <TableCell key={i} className="px-2 py-1 text-center font-mono text-xs tabular-nums">
   109→          {b}
   110→        </TableCell>
   111→      ))}
   112→      <TableCell className="px-2 py-1 text-center font-mono text-xs tabular-nums">
   113→        {row.input}
   114→      </TableCell>
   115→      <TableCell className="px-2 py-1 text-center font-mono text-xs">
   116→        {formatHex(row.input)}
   117→      </TableCell>
   118→      <TableCell className="border-r border-border/70 px-2 py-1 text-center font-mono text-xs">
   119→        {char === null ? '' : char === ' ' ? '␠' : char}
   120→      </TableCell>
   121→      <TableCell className="px-1 py-0.5 text-center">
   122→        <div
   123→          className="group relative flex cursor-grab flex-col items-center leading-none active:cursor-grabbing"
   124→          draggable
   125→          onDragStart={(e) => {
   126→            e.dataTransfer.setData('text/x-sevenseg-row', String(row.input));
   127→            e.dataTransfer.effectAllowed = 'copy';
   128→            onDragRowStart(row.input);
   129→          }}
   130→          onDragEnd={() => {
   131→            onDragRowEnd();
   132→          }}
   133→          title={`Pattern of m${row.input} — click a segment to toggle it (0 ↔ 1); drag onto another row to copy this pattern (inputs are excluded and stay unchanged)`}
   134→        >
   135→          <GlyphPreview
   136→            outputs={row.outputs}
   137→            interactive
   138→            onSegmentClick={(seg) => {
   139→              onTogglePreviewSegment(row.input, seg);
   140→            }}
   141→            className="relative z-0 h-10 origin-center transition-transform duration-150 ease-out group-hover:z-20 group-hover:scale-[1.65]"
   142→            label={`Pattern of row ${row.input}${char ? ` (${char})` : ''} — segments clickable`}
   143→          />
   144→        </div>
   145→      </TableCell>
   146→      {SEGMENT_NAMES.map((seg) => {
   147→        const v = row.outputs[seg];
   148→        return (
   149→          <TableCell
   150→            key={seg}
   151→            className={cn('px-1 py-1 text-center', seg === 'DP' && 'border-l border-border/70')}
   152→          >
   153→            <button
   154→              type="button"
   155→              className={cn(
   156→                'inline-flex h-7 w-7 items-center justify-center rounded border font-mono text-xs font-semibold transition-all hover:scale-110 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring',
   157→                cellColor(v),
   158→              )}
   159→              onClick={(e) => {
   160→                e.stopPropagation();
   161→                onCycleCell(row.input, seg);
   162→              }}
   163→              aria-label={`Row ${row.input} segment ${seg}: ${v}, click to cycle`}
   164→              title={`m${row.input} · ${seg} = ${v} — click to cycle`}
   165→            >
   166→              {v}
   167→            </button>
   168→          </TableCell>
   169→        );
   170→      })}
   171→    </TableRow>
   172→  );
   173→});
