   168→function persist(state: SimulatorState) {
   169→  if (typeof window === 'undefined') return;
   170→  try {
   171→    const data = state.toProjectData();
   172→    window.localStorage.setItem(WORKSPACE_KEY, JSON.stringify(data));
   173→  } catch {
   174→    /* storage full / unavailable — demo continues in memory */
   175→  }
   176→}
   177→
   178→/** Write the preset library to localStorage. */
   179→function persistLibrary(library: PresetLibrary) {
   180→  if (typeof window === 'undefined') return;
   181→  try {
   182→    window.localStorage.setItem(PRESET_LIB_KEY, JSON.stringify(library));
   183→  } catch {
   184→    /* storage full — library stays in memory for this session */
   185→  }
   186→}
   187→
   188→/** Load + validate the preset library from localStorage. */
   189→function loadLibrary(): PresetLibrary {
   190→  if (typeof window === 'undefined') return { tables: [], fonts: [] };
   191→  try {
   192→    const raw = window.localStorage.getItem(PRESET_LIB_KEY);
   193→    if (!raw) return { tables: [], fonts: [] };
   194→    const parsed = JSON.parse(raw) as { tables?: unknown; fonts?: unknown };
   195→    const tables = Array.isArray(parsed.tables)
   196→      ? parsed.tables.map(validateLibraryTablePreset).filter((t): t is TablePresetDef => t !== null)
   197→      : [];
   198→    const fonts = Array.isArray(parsed.fonts)
   199→      ? parsed.fonts.map(validateLibraryFontPreset).filter((f): f is FontPresetDef => f !== null)
   200→      : [];
   201→    return { tables, fonts };
   202→  } catch {
   203→    return { tables: [], fonts: [] };
   204→  }
   205→}

   207→/** Best-effort server sync of one library preset (fire and forget). */
   208→async function apiUpsertPreset(def: TablePresetDef | FontPresetDef, kind: PresetKind): Promise<void> {
   209→  try {
   210→    const res = await fetch('/api/presets', {
   211→      method: 'POST',
   212→      headers: { 'Content-Type': 'application/json' },
   213→      body: JSON.stringify({ id: def.id, kind, name: def.name, data: def }),
   214→    });
   215→    if (res.status === 409) {
   216→      await fetch(`/api/presets/${def.id}`, {
   217→        method: 'PUT',
   218→        headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify({ name: def.name, data: def }),
   220→      });
   221→    }
   222→  } catch {
   223→    /* offline — localStorage remains the source of truth */
   224→  }
   225→}

   227→/** Resolve a table preset by id across built-ins and the user library. */
   228→export function resolveTablePreset(id: string | null, library: PresetLibrary): TablePresetDef | null {
   229→  if (!id) return null;
   230→  return findBuiltinTablePreset(id) ?? library.tables.find((t) => t.id === id) ?? null;
   231→}
   232→
   233→/** Resolve a font preset by id across built-ins and the user library. */
   234→export function resolveFontPreset(id: string, library: PresetLibrary): FontPresetDef | null {
   235→  const builtin = findBuiltinFont(id);
   236→  if (builtin) return builtin;
   237→  return library.fonts.find((f) => f.id === id) ?? null;
   238→}

   240→/** Active font glyph list: builtin ids select builtin fonts; otherwise customFont wins. */
   241→export function activeFontGlyphs(customFont: CustomFont | null, activeFontId = 'builtin:chart'): FontGlyph[] {
   242→  if (activeFontId === 'builtin:calculator') return CALCULATOR_GLYPHS as FontGlyph[];
   243→  if (activeFontId === 'builtin:chart') return FONT_GLYPHS;
   244→  return customFont?.glyphs ?? FONT_GLYPHS;
   245→}

   250→/** Synthesize the builtin table preset id/name implied by a display mode. */
   251→function builtinForMode(mode: DisplayMode): { id: string; name: string } | null {
   252→  if (mode === 'decimal') {
   253→    const p = findBuiltinTablePreset('builtin:bcd')!;
   254→    return { id: p.id, name: p.name };
   255→  }
   256→  if (mode === 'hex') {
   257→    const p = findBuiltinTablePreset('builtin:hex')!;
   258→    return { id: p.id, name: p.name };
   259→  }
   260→  if (mode === 'font') {
   261→    const p = findBuiltinTablePreset('builtin:font')!;
   262→    return { id: p.id, name: p.name };
   263→  }
   264→  return null;
   265→}

   267→/**
   268→ * Make customFont / activeFontId / library consistent:
   269→ * a user font id must have a library entry AND a customFont blob.
   270→ * Returns the patch to apply (empty object when already consistent).
   271→ */
   272→function reconcileFont(
   273→  customFont: CustomFont | null,
   274→  activeFontId: string,
   275→  library: PresetLibrary,
   276→): { customFont?: CustomFont | null; activeFontId?: string; library?: PresetLibrary } {
   277→  if (activeFontId === 'builtin:chart' || activeFontId === 'builtin:calculator') {
   278→    return customFont ? { customFont: null } : {};
   279→  }
   280→  // user font id (or legacy sentinel)
   281→  const entry = library.fonts.find((f) => f.id === activeFontId);
   282→  if (entry) {
   283→    if (customFont && customFont.name === entry.name
   284→      && JSON.stringify(customFont.glyphs) === JSON.stringify(entry.glyphs)) return {};
   285→    return { customFont: { name: entry.name, glyphs: entry.glyphs } };
   286→  }
   287→  // not in library — synthesize from customFont when possible
   288→  if (customFont) {
   289→    const id = activeFontId === FONT_CUSTOM_SENTINEL ? newUserId() : activeFontId;
   290→    const def: FontPresetDef = {
   291→      id,
   292→      name: customFont.name,
   293→      builtin: false,
   294→      glyphs: customFont.glyphs,
   295→      createdAt: new Date().toISOString(),
   296→      updatedAt: new Date().toISOString(),
   297→    };
   298→    return {
   299→      activeFontId: id,
   300→      library: { tables: library.tables, fonts: [...library.fonts, def] },
   301→    };
   302→  }
   303→  return { activeFontId: 'builtin:chart' };
   304→}

   456→  setInputBits: (bits) => {
   457→    if (bits === get().inputBits) return;
   458→    const fontMode = get().displayMode === 'font';
   459→    const applied = resolveTablePreset(get().appliedPresetId, get().presetLibrary);
   460→    const regen = applied && ['excess3', 'gray', 'aiken', 'allon', 'alloff'].includes(applied.generator);
   461→    let table: TruthTable;
   462→    if (fontMode) {
   463→      table = presetTable(bits, 'font', get().invalidBehavior, undefined, {
   464→        fontStart: get().fontStart,
   465→        tableStartCode: get().tableStartCode,
   466→        glyphs: activeFontGlyphs(get().customFont, get().activeFontId),
   467→      });
   468→    } else if (regen && applied) {
   469→      table = buildPresetTable(applied, bits, get().invalidBehavior, activeFontGlyphs(get().customFont, get().activeFontId));
   470→    } else {
   471→      const resized = resizeTable(get().truthTable, bits);
   472→      table = applyInvalidBehavior(resized, get().displayMode, get().invalidBehavior);
   473→    }
   474→    const max = maxvalue(bits);
   475→    set({
   476→      inputBits: bits,
   477→      truthTable: table,
   478→      currentInput: Math.min(get().currentInput, max),
   479→      tableStartCode: Math.min(get().tableStartCode, max),
   480→      undoSnapshot: null,
   481→      lastAction: `Switched to ${bits}-bit input (${bits === 5 ? 'E D C B A' : 'D C B A'})`,
   482→    });
   483→    persist(get());
   484→  },
   485→
   486→  setInput: (value) => {
   487→    const max = maxvalue(get().inputBits);
   488→    const v = Math.min(Math.max(0, Math.round(value)), max);
   489→    if (v !== get().currentInput) set({ currentInput: v });
   490→  },
   491→
   492→  toggleBit: (msbIndex) => {
   493→    const { currentInput, inputBits } = get();
   494→    const bitPos = inputBits - 1 - msbIndex; // convert MSB index to weight
   495→    const next = currentInput ^ (1 << bitPos);
   496→    set({ currentInput: next });
   497→  },
   498→
   499→  incrementInput: () => {
   500→    const { currentInput, inputBits } = get();
   501→    const max = maxvalue(inputBits);
   502→    set({ currentInput: currentInput >= max ? 0 : currentInput + 1 });
   503→  },
   504→
   505→  decrementInput: () => {
   506→    const { currentInput, inputBits } = get();
   507→    const max = maxvalue(inputBits);
   508→    set({ currentInput: currentInput <= 0 ? max : currentInput - 1 });
   509→  },
   510→
   511→  setDisplayMode: (mode) => {
   512→    const prev = get().displayMode;
   513→    if (prev === mode) return;
   514→    const table = mode === 'font'
   515→      ? presetTable(get().inputBits, 'font', get().invalidBehavior, undefined, {
   516→          fontStart: get().fontStart,
   517→          tableStartCode: get().tableStartCode,
   518→          glyphs: activeFontGlyphs(get().customFont, get().activeFontId),
   519→        })
   520→      : presetTable(get().inputBits, mode, get().invalidBehavior, get().truthTable);
   521→    const implied = builtinForMode(mode);
   522→    set({
   523→      displayMode: mode,
   524→      truthTable: table,
   525→      coverExcluded: mode === 'font' || mode === 'decimal' || mode === 'hex' ? {} : get().coverExcluded,
   526→      undoSnapshot: null,
   527→      appliedPresetId: implied?.id ?? null,
   528→      appliedPresetName: implied?.name ?? null,
   529→      lastAction:

   908→  syncPresetsFromServer: async () => {
   909→    if (typeof window === 'undefined') return;
   910→    // wait for hydration so the local library is loaded first
   911→    for (let i = 0; i < 100 && !useSimulator.getState().hydrated; i++) {
   912→      await new Promise((r) => setTimeout(r, 50));
   913→    }
   914→    let rows: { id: string; kind: string; data: unknown; updatedAt?: string }[] = [];
   915→    try {
   916→      const res = await fetch('/api/presets?full=1');
   917→      if (!res.ok) return;
   918→      rows = await res.json();
   919→    } catch {
   920→      return; // offline — the local library stands
   921→    }
   922→    if (!Array.isArray(rows)) return;
   923→    const lib = useSimulator.getState().presetLibrary;
   924→    let tables = [...lib.tables];
   925→    let fonts = [...lib.fonts];
   926→    let changed = false;
   927→    const seen = new Set<string>();
   928→    for (const row of rows) {
   929→      if (!row || typeof row.id !== 'string') continue;
   930→      if (row.kind !== 'table' && row.kind !== 'font') continue;
   931→      seen.add(row.id);
   932→      if (row.kind === 'table') {
   933→        const def = validateLibraryTablePreset(row.data);
   934→        if (!def) continue;
   935→        const existing = tables.find((t) => t.id === row.id);
   936→        if (!existing) {
   937→          tables.push(def);
   938→          changed = true;
   939→        } else if ((existing.updatedAt ?? '') < (def.updatedAt ?? row.updatedAt ?? '')
   940→          && JSON.stringify(existing) !== JSON.stringify(def)) {
   941→          tables = tables.map((t) => (t.id === row.id ? def : t));
   942→          changed = true;
   943→        }
   944→      } else {
   945→        const def = validateLibraryFontPreset(row.data);
   946→        if (!def) continue;
   947→        const existing = fonts.find((f) => f.id === row.id);
   948→        if (!existing) {
   949→          fonts.push(def);
   950→          changed = true;
   951→        } else if ((existing.updatedAt ?? '') < (def.updatedAt ?? row.updatedAt ?? '')
   952→          && JSON.stringify(existing) !== JSON.stringify(def)) {
   953→          fonts = fonts.map((f) => (f.id === row.id ? def : f));
   954→          changed = true;
   955→        }
   956→      }
   957→    }
   958→    // push local-only presets up to the server
   959→    for (const t of tables) if (!seen.has(t.id)) void apiUpsertPreset(t, 'table');
   960→    for (const f of fonts) if (!seen.has(f.id)) void apiUpsertPreset(f, 'font');
   961→    if (changed) {
   962→      const library = { tables, fonts };
   963→      useSimulator.setState({ presetLibrary: library });
   964→      persistLibrary(library);
   965→    }
   966→  },

  1100→/** Subscribe once from a client component to persist every change (debounced). */
  1101→export function attachPersistence() {
  1102→  if (typeof window === 'undefined') return () => {};
  1103→  let timer: ReturnType<typeof setTimeout> | null = null;
  1104→  const unsub = useSimulator.subscribe(() => {
  1105→    if (timer) clearTimeout(timer);
  1106→    timer = setTimeout(() => persist(useSimulator.getState()), 250);
  1107→  });
  1108→  return unsub;
